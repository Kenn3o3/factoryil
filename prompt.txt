def vec2mat(vec):
    x, y = vec[..., [0], :], vec[..., [1], :]
    x_norm = F.normalize(x, dim=-1)
    y = y - x_norm * torch.sum(x_norm * y, dim=-1, keepdim=True)
    y_norm = F.normalize(y, dim=-1)
    z = torch.cross(x_norm, y_norm, dim=-1)
    mat = torch.cat([x_norm, y_norm, z], dim=-2).transpose(-2, -1)
    return mat


class EquiBotPolicy(nn.Module):
    def __init__(self, cfg, device="cpu"):
        nn.Module.__init__(self)
        self.obs_mode = cfg.model.obs_mode
        self.ac_mode = cfg.model.ac_mode

        self.use_torch_compile = cfg.model.use_torch_compile
        self.device = device

        self.pred_horizon = cfg.model.pred_horizon
        self.obs_horizon = cfg.model.obs_horizon
        self.action_horizon = cfg.model.ac_horizon

        if hasattr(cfg.model, "num_diffusion_iters"):
            self.num_diffusion_iters = cfg.model.num_diffusion_iters
        else:
            self.num_diffusion_iters = cfg.model.noise_scheduler.num_train_timesteps

        if self.obs_mode.startswith("pc"):
            self.encoder = SIM3Vec4Latent(**cfg.model.encoder)
        else:
            self.encoder = None
        self.encoder_out_dim = cfg.model.encoder.c_dim

        self.num_eef = cfg.env.num_eef
        self.eef_dim = cfg.env.eef_dim
        self.dof = cfg.env.dof

        if cfg.model.obs_mode == "state":

            self.obs_dim = self.num_eef * (self.eef_dim // 3)
        elif cfg.model.obs_mode == "rgb":
            raise NotImplementedError()
        else:

            self.obs_dim = self.encoder_out_dim + self.num_eef * (self.eef_dim // 3)

        self.action_dim = (2 if self.dof > 4 else 1) * self.num_eef

        num_scalar_dims = (0 if self.dof == 3 else 1) * self.num_eef

        self.noise_pred_net = VecConditionalUnet1D(
            input_dim=self.action_dim,
            cond_dim=self.obs_dim * self.obs_horizon,
            scalar_cond_dim=(0 if self.dof == 3 else self.num_eef * self.obs_horizon),
            scalar_input_dim=num_scalar_dims,
            diffusion_step_embed_dim=self.obs_dim * self.obs_horizon,
            cond_predict_scale=True,
        )

        self.nets = nn.ModuleDict(
            {"encoder": self.encoder, "noise_pred_net": self.noise_pred_net}
        )
        self.ema = EMAModel(model=copy.deepcopy(self.nets), power=0.75)

        self._init_torch_compile()

        self.noise_scheduler = hydra.utils.instantiate(cfg.model.noise_scheduler)

        num_parameters = sum(p.numel() for p in self.parameters() if p.requires_grad)
        print(f"Initialized SIM(3) DP Policy with {num_parameters} parameters")

    def _init_torch_compile(self):
        if self.use_torch_compile:
            self.encoder_handle = torch.compile(self.encoder)
            self.noise_pred_net_handle = torch.compile(self.noise_pred_net)
        else:
            self.encoder_handle = self.encoder
            self.noise_pred_net_handle = self.noise_pred_net

    def _convert_state_to_vec(self, state):
        if self.dof == 3:
            return state.view(state.shape[0], state.shape[1], -1, 3), None, None
        elif self.dof == 4:
            state = state.view(state.shape[0], state.shape[1], self.num_eef, -1)
            assert state.shape[-1] in [4, 7]
            eef_pose_w = state[:, :, :, :3]
            scalar_state = state[:, :, :, 3]
            if state.shape[-1] == 7:
                goal_pos = state[:, :, :, -3:]
                vec_state_pos = torch.cat([eef_pose_w, goal_pos], dim=2)
            else:
                vec_state_pos = eef_pose_w
            return vec_state_pos, None, scalar_state
        else:
            state = state.view(state.shape[0], state.shape[1], self.num_eef, -1)

            assert state.shape[-1] in [10, 13, 16]
            eef_pose_w = state[:, :, :, :3]
            dir1 = state[:, :, :, 3:6]
            dir2 = state[:, :, :, 6:9]
            gripper_pose = state[:, :, :, 9]

            vec_state_pos = eef_pose_w
            vec_state_dir = torch.cat([dir1, dir2], dim=2)
            scalar_state = gripper_pose
            return vec_state_pos, vec_state_dir, scalar_state

    def _convert_action_to_vec(self, ac, batch=None):

        ac = ac.view(ac.shape[0], ac.shape[1], -1, self.dof)
        if self.dof == 4:

            gripper_ac = ac[:, :, :, 3]
            eef_ac = ac[:, :, :, 0:3]
            return eef_ac.permute(0, 2, 3, 1), gripper_ac.permute(0, 2, 1)
        elif self.dof == 7:

            gripper_ac = ac[:, :, :, 6]
            eef_ac = ac[:, :, :, 0:6]
            eef_ac = eef_ac.reshape(ac.shape[0], ac.shape[1], -1, 3)
            return eef_ac.permute(0, 2, 3, 1), gripper_ac.permute(0, 2, 1)
        elif self.dof == 3:
            return ac.permute(0, 2, 3, 1), None
        else:
            raise ValueError(f"Cannot handle dof = {self.dof}")

    def _convert_action_to_scalar(self, eef_ac, gripper_ac=None, batch=None):

        if self.dof in [4, 7]:
            assert len(eef_ac.shape) == 4
            assert len(gripper_ac.shape) == 3
            assert eef_ac.shape[0] == gripper_ac.shape[0]
            assert eef_ac.shape[-1] == gripper_ac.shape[-1]
            assert eef_ac.shape[2] == 3
            eef_ac = eef_ac.reshape(
                eef_ac.shape[0], self.num_eef, 3 if self.dof == 4 else 6, -1
            )
            grip_ac = gripper_ac[:, :, None]
            scalar_ac = torch.cat([eef_ac, grip_ac], dim=2).permute(0, 3, 1, 2)
            scalar_ac = scalar_ac.reshape(scalar_ac.shape[0], scalar_ac.shape[1], -1)
        elif self.dof == 3:
            scalar_ac = eef_ac.reshape(eef_ac.shape[0], -1, eef_ac.shape[-1]).permute(
                0, 2, 1
            )
        else:
            raise ValueError(f"Cannot handle dof = {self.dof}")
        return scalar_ac

    def forward(self, obs, predict_action=True, debug=False):

        pc = obs["pc"]
        state = obs["state"]

        pc = self.pc_normalizer.normalize(pc)

        pc_shape = pc.shape
        batch_size = B = pc.shape[0]
        Ho = self.obs_horizon
        Hp = self.pred_horizon

        ema_nets = self.ema.averaged_model

        if self.obs_mode == "state":
            z_pos, z_dir, z_scalar = self._convert_state_to_vec(state)
            z_pos = self.state_normalizer.normalize(z_pos)
            if self.dof > 4:
                z = torch.cat([z_pos, z_dir], dim=-2)
            else:
                z = z_pos
        else:
            feat_dict = ema_nets["encoder"](pc, target_norm=self.pc_scale)
            center = (
                feat_dict["center"].reshape(B, Ho, 1, 3)[:, [-1]].repeat(1, Ho, 1, 1)
            )
            scale = feat_dict["scale"].reshape(B, Ho, 1, 1)[:, [-1]].repeat(1, Ho, 1, 1)
            z_pos, z_dir, z_scalar = self._convert_state_to_vec(state)
            z_pos = self.state_normalizer.normalize(z_pos)
            z_pos = (z_pos - center) / scale
            z = feat_dict["so3"]
            z = z.reshape(B, Ho, -1, 3)
            if self.dof == 7:
                z = torch.cat([z, z_pos, z_dir], dim=-2)
            else:
                z = torch.cat([z, z_pos], dim=-2)
        obs_cond_vec, obs_cond_scalar = z.reshape(B, -1, 3), (
            z_scalar.reshape(B, -1) if z_scalar is not None else None
        )

        initial_noise_scale = 0.0 if debug else 1.0
        if self.dof > 3:
            noisy_action = (
                torch.randn((B, Hp, self.action_dim, 3)).to(self.device)
                * initial_noise_scale,
                torch.randn((B, Hp, self.num_eef)).to(self.device)
                * initial_noise_scale,
            )
        else:
            noisy_action = (
                torch.randn((B, Hp, self.action_dim, 3)).to(self.device)
                * initial_noise_scale,
                None,
            )
        curr_action = noisy_action
        self.noise_scheduler.set_timesteps(self.num_diffusion_iters)

        for k in self.noise_scheduler.timesteps:

            noise_pred = ema_nets["noise_pred_net"](
                sample=curr_action[0],
                timestep=k,
                scalar_sample=curr_action[1],
                cond=obs_cond_vec,
                scalar_cond=obs_cond_scalar,
            )

            new_action = [None, None]
            new_action[0] = self.noise_scheduler.step(
                model_output=noise_pred[0], timestep=k, sample=curr_action[0]
            ).prev_sample
            if noise_pred[1] is not None:
                new_action[1] = self.noise_scheduler.step(
                    model_output=noise_pred[1], timestep=k, sample=curr_action[1]
                ).prev_sample
            curr_action = tuple(new_action)

        if curr_action[1] is not None:
            action = self._convert_action_to_scalar(
                curr_action[0].permute(0, 2, 3, 1),
                curr_action[1].permute(0, 2, 1),
                batch=obs,
            )
        else:
            action = self._convert_action_to_scalar(
                curr_action[0].permute(0, 2, 3, 1), batch=obs
            )

        ret = dict(ac=action)
        if debug:
            ret.update(
                dict(
                    obs_cond_vec=obs_cond_vec.detach().cpu().numpy(),
                )
            )
            if obs_cond_scalar is not None:
                ret.update(
                    dict(
                        obs_cond_scalar=(
                            obs_cond_scalar.detach().cpu().numpy()
                            if obs_cond_scalar is not None
                            else None
                        )
                    )
                )
            if self.obs_mode != "state":
                ret.update(
                    dict(
                        center=feat_dict["center"]
                        .detach()
                        .reshape(B, Ho, 3)
                        .cpu()
                        .numpy(),
                        scale=feat_dict["scale"].detach().reshape(B, Ho).cpu().numpy(),
                        feat_so3=feat_dict["so3"]
                        .detach()
                        .reshape(B, Ho, -1, 3)
                        .cpu()
                        .numpy(),
                    )
                )
        return ret

    def step_ema(self):
        self.ema.step(self.nets)

import torch
from torch import nn
import numpy as np
import time

from ..utils.norm import Normalizer
from ..utils.misc import to_torch
from .dp_agent import DPAgent
from .equibot_policy import EquiBotPolicy


class EquiBotAgent(DPAgent):
    def _init_actor(self):
        self.actor = EquiBotPolicy(self.cfg, device=self.cfg.device).to(self.cfg.device)
        self.actor.ema.averaged_model.to(self.cfg.device)

        self.pc_scale = None

    def _init_normalizers(self, batch):
        if self.ac_normalizer is None:
            gt_action = batch["action"]

            flattened_gt_action = gt_action.view(-1, self.dof)

            if self.dof == 7:
                indices = [[6], [0, 1, 2], [3, 4, 5]]
            elif self.dof == 4:
                indices = [[3], [0, 1, 2]]
            else:
                indices = None

            ac_normalizer = Normalizer(
                flattened_gt_action, symmetric=True, indices=indices
            )

            self.ac_normalizer = Normalizer(
                {
                    "min": ac_normalizer.stats["min"].tile((self.num_eef,)),
                    "max": ac_normalizer.stats["max"].tile((self.num_eef,)),
                }
            )
            print(f"Action normalization stats: {self.ac_normalizer.stats}")
        if self.state_normalizer is None:

            if self.dof == 3:

                self.state_normalizer = ac_normalizer
            else:

                self.state_normalizer = Normalizer(
                    {
                        "min": ac_normalizer.stats["min"][:3],
                        "max": ac_normalizer.stats["max"][:3],
                    }
                )

            self.actor.state_normalizer = self.state_normalizer

        if self.pc_normalizer is None:

            self.pc_normalizer = self.state_normalizer
            self.actor.pc_normalizer = self.pc_normalizer

        pc = batch["pc"].reshape(-1, self.num_points, 3)
        centroid = pc.mean(1, keepdim=True)
        centered_pc = pc - centroid
        pc_scale = centered_pc.norm(dim=-1).mean()

        ac_scale = ac_normalizer.stats["max"].max()

        self.pc_scale = pc_scale / ac_scale
        self.actor.pc_scale = self.pc_scale

    def update(self, batch, training=True):
        self.train(training)

        batch = to_torch(batch, self.device)
        pc = batch["pc"]
        state = batch["eef_pose_w"]
        gt_action = batch["action"]

        if self.pc_scale is None:
            self._init_normalizers(batch)

        pc = self.pc_normalizer.normalize(pc)
        gt_action = self.ac_normalizer.normalize(gt_action)

        pc_shape = pc.shape
        batch_size = B = pc.shape[0]
        Ho = self.obs_horizon
        Hp = self.pred_horizon

        if self.obs_mode == "state":

            z_pos, z_dir, z_scalar = self.actor._convert_state_to_vec(state)
            z_pos = self.state_normalizer.normalize(z_pos)
            if self.dof > 4:

                z = torch.cat([z_pos, z_dir], dim=-2)
            else:
                z = z_pos
        else:

            feat_dict = self.actor.encoder_handle(pc, target_norm=self.pc_scale)

            center = (
                feat_dict["center"].reshape(B, Ho, 1, 3)[:, [-1]].repeat(1, Ho, 1, 1)
            )
            scale = feat_dict["scale"].reshape(B, Ho, 1, 1)[:, [-1]].repeat(1, Ho, 1, 1)

            z_pos, z_dir, z_scalar = self.actor._convert_state_to_vec(state)
            z_pos = self.state_normalizer.normalize(z_pos)
            z_pos = (z_pos - center) / scale

            z = feat_dict["so3"]
            z = z.reshape(B, Ho, -1, 3)
            if self.dof > 4:
                z = torch.cat([z, z_pos, z_dir], dim=-2)
            else:
                z = torch.cat([z, z_pos], dim=-2)

        obs_cond_vec = z.reshape(B, -1, 3)
        obs_cond_scalar = z_scalar.reshape(B, -1) if z_scalar is not None else None

        if self.obs_mode.startswith("pc") and getattr(
            self.actor, "use_local_action_frame", True
        ):

            if self.ac_mode == "abs":
                center = (
                    feat_dict["center"]
                    .reshape(B, Ho, 1, 3)[:, [-1]]
                    .repeat(1, Hp, 1, 1)
                )
            else:
                center = 0
            scale = feat_dict["scale"].reshape(B, Ho, 1, 1)[:, [-1]].repeat(1, Hp, 1, 1)
            gt_action = gt_action.reshape(B, Hp, self.num_eef, self.dof)
            if self.dof == 4:

                gt_action = torch.cat(
                    [(gt_action[..., 0:3] - center) / scale, gt_action[..., 3:4]],
                    dim=-1,
                )
                gt_action = gt_action.reshape(B, Hp, -1)
            elif self.dof == 3:

                gt_action = (gt_action - center) / scale
            elif self.dof == 7:

                gt_action = torch.cat(
                    [
                        (gt_action[..., 0:3] - center) / scale,
                        gt_action[..., 3:6],
                        gt_action[..., 6:7],
                    ],
                    dim=-1,
                )
                gt_action = gt_action.reshape(B, Hp, -1)
            else:
                raise ValueError(f"Dof {self.dof} not supported.")

        vec_eef_action, vec_gripper_action = self.actor._convert_action_to_vec(
            gt_action, batch
        )

        if self.dof != 7:
            noise = torch.randn(gt_action.shape, device=self.device)
            vec_eef_noise, vec_gripper_noise = self.actor._convert_action_to_vec(
                noise, batch
            )
        else:
            vec_eef_noise = torch.randn_like(vec_eef_action, device=self.device)
            vec_gripper_noise = torch.randn_like(vec_gripper_action, device=self.device)

        timesteps = torch.randint(
            0,
            self.actor.noise_scheduler.config.num_train_timesteps,
            (B,),
            device=self.device,
        ).long()

        if vec_gripper_action is not None:
            noisy_eef_actions = self.actor.noise_scheduler.add_noise(
                vec_eef_action, vec_eef_noise, timesteps
            )
            noisy_gripper_actions = self.actor.noise_scheduler.add_noise(
                vec_gripper_action, vec_gripper_noise, timesteps
            )

            vec_eef_noise_pred, vec_gripper_noise_pred = (
                self.actor.noise_pred_net_handle(
                    noisy_eef_actions.permute(0, 3, 1, 2),
                    timesteps,
                    scalar_sample=noisy_gripper_actions.permute(0, 2, 1),
                    cond=obs_cond_vec,
                    scalar_cond=obs_cond_scalar,
                )
            )
            vec_eef_noise_pred = vec_eef_noise_pred.permute(0, 2, 3, 1)
            vec_gripper_noise_pred = vec_gripper_noise_pred.permute(0, 2, 1)
            if self.dof != 7:

                noise_pred = self.actor._convert_action_to_scalar(
                    vec_eef_noise_pred, vec_gripper_noise_pred, batch=batch
                ).view(noise.shape)
        else:
            noisy_eef_actions = self.actor.noise_scheduler.add_noise(
                vec_eef_action, vec_eef_noise, timesteps
            )

            vec_noise_pred = self.actor.noise_pred_net_handle(
                noisy_eef_actions.permute(0, 3, 1, 2),
                timesteps,
                cond=obs_cond_vec,
                scalar_cond=obs_cond_scalar,
            )[0].permute(0, 2, 3, 1)
            if self.dof != 7:
                noise_pred = self.actor._convert_action_to_scalar(
                    vec_noise_pred, batch=batch
                ).view(noise.shape)

        if self.dof == 7:
            n_vec = np.prod(vec_eef_noise_pred.shape)
            n_sca = np.prod(vec_gripper_noise_pred.shape)
            k = (n_vec) / (n_vec + n_sca)
            loss = nn.functional.mse_loss(
                vec_eef_noise_pred, vec_eef_noise
            ) * k + nn.functional.mse_loss(
                vec_gripper_noise_pred, vec_gripper_noise
            ) * (
                1 - k
            )
        else:
            loss = nn.functional.mse_loss(noise_pred, noise)
        if torch.isnan(loss):
            print(f"Loss is nan, please investigate.")
            import pdb

            pdb.set_trace()

        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        self.lr_scheduler.step()

        self.actor.step_ema()

        metrics = {
            "loss": loss,
            "normalized_gt_ac_max": np.max(
                np.abs(vec_eef_action.reshape(-1, 3).detach().cpu().numpy()), axis=0
            ).mean(),
        }
        if self.dof == 7:
            metrics.update(
                {
                    "mean_gt_eef_noise_norm": np.linalg.norm(
                        vec_eef_noise.detach().cpu().numpy(), axis=1
                    ).mean(),
                    "mean_pred_eef_noise_norm": np.linalg.norm(
                        vec_eef_noise_pred.detach().cpu().numpy(), axis=1
                    ).mean(),
                    "mean_gt_gripper_noise_norm": np.linalg.norm(
                        vec_gripper_noise.detach().cpu().numpy(), axis=1
                    ).mean(),
                    "mean_pred_gripper_noise_norm": np.linalg.norm(
                        vec_gripper_noise_pred.detach().cpu().numpy(), axis=1
                    ).mean(),
                }
            )
        else:
            metrics.update(
                {
                    "mean_gt_noise_norm": np.linalg.norm(
                        noise.reshape(gt_action.shape[0], -1).detach().cpu().numpy(),
                        axis=1,
                    ).mean(),
                    "mean_pred_noise_norm": np.linalg.norm(
                        noise_pred.reshape(gt_action.shape[0], -1)
                        .detach()
                        .cpu()
                        .numpy(),
                        axis=1,
                    ).mean(),
                }
            )

        return metrics

                vec_eef_noise_pred, vec_gripper_noise_pred = self.actor.noise_pred_net(
                    noisy_eef_actions.permute(0, 3, 1, 2),
                    timesteps,
                    scalar_sample=noisy_gripper_actions.permute(0, 2, 1),
                    cond=obs_cond_vec,
                    scalar_cond=obs_cond_scalar,
                )

                vec_gripper_noise_pred = vec_gripper_noise_pred.permute(0, 2, 1)

                if self.dof != 7:
                    noise_pred = self.actor._convert_action_to_scalar(
                        vec_eef_noise_pred, vec_gripper_noise_pred, batch=batch
                    ).view(noise.shape)
            else:

                    vec_eef_action, vec_eef_noise, timesteps
                )
                vec_noise_pred = self.actor.noise_pred_net(
                    noisy_eef_actions.permute(0, 3, 1, 2),
                    timesteps,
                    cond=obs_cond_vec,
                    scalar_cond=obs_cond_scalar,
                )[0].permute(0, 2, 3, 1)
                if self.dof != 7:
                    noise_pred = self.actor._convert_action_to_scalar(
                        vec_noise_pred, batch=batch
                    ).view(noise.shape)

                n_sca = int(np.prod(vec_gripper_noise_pred.shape))
                k = n_vec / (n_vec + n_sca)
                loss = nn.functional.mse_loss(
                    vec_eef_noise_pred, vec_eef_noise
                ) * k + nn.functional.mse_loss(
                    vec_gripper_noise_pred, vec_gripper_noise

                    1 - k
                )

                gt_flat = torch.cat(
                    [vec_eef_noise.reshape(B, -1), vec_gripper_noise.reshape(B, -1)],
                    dim=1,
                )
                pr_flat = torch.cat(
                    [
                        vec_eef_noise_pred.reshape(B, -1),
                        vec_gripper_noise_pred.reshape(B, -1),
                    ],
                    dim=1,
                )
            else:
                loss = nn.functional.mse_loss(noise_pred, noise)
                gt_flat = noise.reshape(B, -1)
                pr_flat = noise_pred.reshape(B, -1)

            metrics = {
                "loss": float(loss.detach().cpu().item()),
                "mean_gt_noise_norm": float(
                    torch.linalg.norm(gt_flat, dim=1).mean().detach().cpu().item()
                ),
                "mean_pred_noise_norm": float(
                    torch.linalg.norm(pr_flat, dim=1).mean().detach().cpu().item()
                ),
            }
            return metrics

BACKBONE_DICT = {"vn_pointnet": VecPointNet}

class SIM3Vec4Latent(nn.Module):
    def __init__(
        self,
        c_dim,
        backbone_type,
        backbone_args,
        mode="so3",
        normalization_method=None,
    ):
        super().__init__()
        assert mode == "so3", NotImplementedError("TODO, add se3")
        if normalization_method is not None:
            backbone_args["normalization_method"] = NORMALIZATION_METHOD[
                normalization_method
            ]

        self.backbone = BACKBONE_DICT[backbone_type](**backbone_args)

        self.fc_inv = VecLinear(c_dim, c_dim, mode=mode)

    def forward(self, pcl, ret_perpoint_feat=False, target_norm=1.0):
        B, T, N, _ = pcl.shape
        pcl = pcl.view(B * T, -1, 3).transpose(1, 2)

        centroid = pcl.mean(-1, keepdim=True)
        input_pcl = pcl - centroid

        z_scale = input_pcl.norm(dim=1).mean(-1) / target_norm
        z_center = centroid.permute(0, 2, 1)

        input_pcl = input_pcl / z_scale[:, None, None]

        x, x_perpoint = self.backbone(input_pcl)

        z_so3 = x
        z_inv_dual, _ = self.fc_inv(x[..., None])
        z_inv_dual = z_inv_dual.squeeze(-1)
        z_inv = (z_inv_dual * z_so3).sum(-1)

        ret = {
            "inv": z_inv,
            "so3": z_so3,
            "scale": z_scale,
            "center": z_center,
        }

        if ret_perpoint_feat:
            ret["per_point_so3"] = x_perpoint

        return ret

import os
import sys
import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
from pytorch3d.ops.knn import knn_points
import logging

from .vec_layers import VecLinear
from .vec_layers import VecLinNormAct as VecLNA


def meanpool(x, dim=-1, keepdim=False):
    out = x.mean(dim=dim, keepdim=keepdim)
    return out


class VecPointNet(nn.Module):
    def __init__(
        self,
        h_dim=128,
        c_dim=128,
        num_layers=4,
        knn=16,
    ):
        super().__init__()

        self.h_dim = h_dim
        self.c_dim = c_dim
        self.num_layers = num_layers
        self.knn = knn

        self.pool = meanpool

        act_func = nn.LeakyReLU(negative_slope=0.0, inplace=False)
        vnla_cfg = {"mode": "so3", "act_func": act_func}

        self.conv_in = VecLNA(3, h_dim, **vnla_cfg)
        self.layers, self.global_layers = nn.ModuleList(), nn.ModuleList()

        for i in range(self.num_layers):

            self.layers.append(VecLNA(h_dim, h_dim, **vnla_cfg))

            self.global_layers.append(VecLNA(h_dim * 2, h_dim, **vnla_cfg))

        self.conv_out = VecLinear(h_dim * self.num_layers, c_dim, mode="so3")

        self.fc_inv = VecLinear(c_dim, 3, mode="so3")

    def get_graph_feature(self, x: torch.Tensor, k: int, knn_idx=None, cross=False):
        B, C, _, N = x.shape
        if knn_idx is None:
            _x = x.reshape(B, -1, N)
            _, knn_idx, neighbors = knn_points(
                _x.transpose(2, 1), _x.transpose(2, 1), K=k, return_nn=True
            )
            neighbors = neighbors.reshape(B, N, k, C, 3).permute(0, -2, -1, 1, 2)
        else:
            assert knn_idx.shape[-1] == k, f"input knn gather idx should have k={k}"
            neighbors = torch.gather(
                x[..., None, :].expand(-1, -1, -1, N, -1),
                dim=-1,
                index=knn_idx[:, None, None, ...].expand(-1, C, 3, -1, -1),
            )
        x_padded = x[..., None].expand_as(neighbors)

        if cross:
            x_dir = F.normalize(x, dim=2)
            x_dir_padded = x_dir[..., None].expand_as(neighbors)
            cross = torch.cross(x_dir_padded, neighbors, dim=2)
            y = torch.cat([cross, neighbors - x_padded, x_padded], 1)
        else:
            y = torch.cat([neighbors - x_padded, x_padded], 1)
        return y, knn_idx

    def forward(self, x):
        x = x.unsqueeze(1)

        x, knn_idx = self.get_graph_feature(x, self.knn, cross=True)
        x, _ = self.conv_in(x)
        x = self.pool(x)

        y = x
        feat_list = []
        for i in range(self.num_layers):
            y, _ = self.layers[i](y)
            y_global = y.mean(-1, keepdim=True)
            y = torch.cat([y, y_global.expand_as(y)], dim=1)
            y, _ = self.global_layers[i](y)
            feat_list.append(y)
        x = torch.cat(feat_list, dim=1)
        x, _ = self.conv_out(x)

        return x.mean(-1), x

import torch
from torch import nn
import torch.nn.functional as F
import math
import logging

EPS = 1e-7

def channel_equi_vec_normalize(x):
    # B,C,3,...
    # 先对每个向量做 L2 方向归一化，再对向量长度在“通道维 C”上做归一化
    assert x.ndim >= 3, "x shape [B,C,3,...]"
    x_dir = F.normalize(x, dim=2)
    x_norm = x.norm(dim=2, keepdim=True)
    x_normalized_norm = F.normalize(x_norm, dim=1)  # normalize across C
    y = x_dir * x_normalized_norm
    return y

class VecLayerNorm(nn.Module):
    def __init__(self, normalized_dim):
        super().__init__()
        self.gamma = nn.Parameter(torch.ones(normalized_dim))

    def forward(self, x):
        x = channel_equi_vec_normalize(x)
        x = x * self.gamma.view(tuple([1, -1] + [1] * (len(x.shape) - 2)))
        return x

class VecLinear(nn.Module):
    r"""
    向量等变的“线性层”，支持 SO(3)/SE(3)，并可混合标量分支。
    模式说明：
    - v_in, v_out: 输入/输出的“向量通道数”（每个通道是一个 3D 向量）
    - s_in, s_out: 可选的标量输入/输出通道数（不随旋转变）
    - mode: "so3"/"se3"（SE(3) 时需要至少两个向量通道以表达平移）
    - cross: 是否使用叉乘增强通道交互
    输出：返回 (向量输出, 标量输出或 None)
    """
    v_in: int
    v_out: int
    s_in: int
    s_out: int
    weight: torch.Tensor

    def __init__(
        self,
        v_in: int,
        v_out: int,
        s_in=0,
        s_out=0,
        mode="so3",
        scalar2vec_normalized_scale=True,
        vec2scalar_dir_learnable=True,
        cross=False,
        device=None,
        dtype=None,
    ) -> None:
        mode = mode.lower()
        assert mode in ["so3", "se3"], "mode must be so3 or se3"
        factory_kwargs = {"device": device, "dtype": dtype}
        super().__init__()

        self.v_in = v_in
        self.v_out = v_out
        self.s_in = s_in
        self.s_out = s_out

        assert self.s_out + self.v_out > 0, "vec, scalar output both zero"

        self.se3_flag = mode == "se3"
        if self.se3_flag:
            assert v_in > 1, "se3 layers must have at least two input layers"

        if self.v_out > 0:
            self.weight = nn.Parameter(
                torch.empty(
                    (v_out, v_in - 1 if self.se3_flag else v_in), **factory_kwargs
                )
            )  # 主向量权重（SE3 模式下约束权重和为 1）
            self.reset_parameters()

        if (
            self.s_in > 0 and self.v_out > 0
        ):  # 有标量输入时，必须提供标量->向量的融合路径
            self.sv_linear = nn.Linear(s_in, v_out)
            self.s2v_normalized_scale_flag = scalar2vec_normalized_scale

        if self.s_out > 0:  # 有标量输出，必须提供向量->标量路径
            self.vec2scalar_dir_learnable = vec2scalar_dir_learnable
            if self.vec2scalar_dir_learnable:
                self.vs_dir_linear = VecLinear(
                    v_in, v_in, mode="so3"
                )  # 可学习的投影方向
            self.vs_linear = nn.Linear(v_in, s_out)
        if self.s_in > 0 and self.s_out > 0:  # 若有 s_in 与 s_out，则存在 s->s 的直连
            self.ss_linear = nn.Linear(s_in, s_out)

        self.cross_flag = cross
        if self.v_out > 0 and self.cross_flag:
            self.v_out_cross = VecLinear(v_in, v_out, mode=mode, cross=False)
            self.v_out_cross_fc = VecLinear(v_out * 2, v_out, mode=mode, cross=False)

    @torch.no_grad()
    def reset_parameters(self) -> None:
        # 简单的 Kaiming 初始化（注意历史权重兼容性）
        nn.init.kaiming_uniform_(self.weight, a=math.sqrt(5))
        if self.se3_flag:
            self.weight.data += 1.0 / self.v_in

    def forward(self, v_input: torch.Tensor, s_input=None):
        # v_input: B,C,3,...
        # s_input: B,C_s,...

        # 向量主路径
        v_shape = v_input.shape
        assert v_shape[2] == 3, "not vector neuron"
        if self.v_out > 0:
            if self.se3_flag:
                W = torch.cat(
                    [self.weight, 1.0 - self.weight.sum(-1, keepdim=True)], -1
                ).contiguous()
            else:
                W = self.weight
            v_output = F.linear(v_input.transpose(1, -1), W).transpose(
                -1, 1
            )  # B,C,3,...
        else:
            v_output = None

        # 标量融合：将 s_input 作用于 v_output 的尺度（FiLM-like）
        if self.s_in > 0:
            assert s_input is not None, "missing scalar input"
            s_shape = s_input.shape
            assert v_shape[3:] == s_shape[2:]
            if self.v_out > 0:
                s2v_invariant_scale = self.sv_linear(
                    s_input.transpose(1, -1)
                ).transpose(-1, 1)
                if self.s2v_normalized_scale_flag:
                    s2v_invariant_scale = F.normalize(s2v_invariant_scale, dim=1)
                if self.se3_flag:  # 仅缩放旋转部分，平移部分不直接缩放
                    v_new_mean = v_output.mean(dim=1, keepdim=True)
                    v_output = (v_output - v_new_mean) * s2v_invariant_scale.unsqueeze(
                        2
                    ) + v_new_mean
                else:
                    v_output = v_output * s2v_invariant_scale.unsqueeze(2)

        # 可选叉乘分支，增强表达（将 dual 通道与输出通道叉乘再融合）
        if self.v_out > 0 and self.cross_flag:
            v_out_dual, _ = self.v_out_cross(v_input)
            if self.se3_flag:
                v_out_dual_o = v_out_dual.mean(dim=1, keepdim=True)
                v_output_o = v_output.mean(dim=1, keepdim=True)
                v_cross = torch.cross(
                    channel_equi_vec_normalize(v_out_dual - v_out_dual_o),
                    v_output - v_output_o,
                    dim=2,
                )
            else:
                v_cross = torch.cross(
                    channel_equi_vec_normalize(v_out_dual), v_output, dim=2
                )
            v_cross = v_cross + v_output
            v_output, _ = self.v_out_cross_fc(torch.cat([v_cross, v_output], dim=1))

        # 向量 -> 标量
        if self.s_out > 0:
            v_sR = (
                v_input - v_input.mean(dim=1, keepdim=True)
                if self.se3_flag
                else v_input
            )
            if self.vec2scalar_dir_learnable:
                v_sR_dual_dir, _ = self.vs_dir_linear(v_sR)
                v_sR_dual_dir = F.normalize(v_sR_dual_dir, dim=2)
                s_from_v = F.normalize(
                    (v_sR * v_sR_dual_dir).sum(dim=2), dim=1
                )  # B,C,...
            else:
                s_from_v = F.normalize(v_sR.norm(dim=2), dim=1)

            s_from_v = self.vs_linear(s_from_v.transpose(-1, 1)).transpose(-1, 1)
            if self.s_in > 0:
                s_from_s = self.ss_linear(s_input.transpose(-1, 1)).transpose(-1, 1)
                s_output = s_from_s + s_from_v
            else:
                s_output = s_from_v
        else:
            s_output = None
        return v_output, s_output

class VecActivation(nn.Module):
    # 向量非线性：
    # - vec 模式：在“与可学习方向 k 的平行分量长度”上施加非线性；正交分量保持
    # - len 模式：对向量长度施加非线性，方向保持
    # 可选：SE(3) 下先减去中心再加回，支持“安全截断”等
    def __init__(
        self,
        in_features,
        act_func,
        shared_nonlinearity=False,
        mode="so3",
        cross=False,
        normalization_method=None,
        safe_bound=-1,
        act_mode="vec",
        pre_act_len_bound=-1.0,
    ) -> None:
        super().__init__()
        self.act_mode = act_mode
        assert self.act_mode in ["vec", "len"], "act_mode must be vec or len"
        mode = mode.lower()
        assert mode in ["so3", "se3"], "mode must be so3 or se3"
        self.se3_flag = mode == "se3"
        self.shared_nonlinearity_flag = shared_nonlinearity
        self.act_func = act_func

        if self.act_mode == "vec":
            nonlinear_out = 1 if self.shared_nonlinearity_flag else in_features
            self.lin_dir = VecLinear(in_features, nonlinear_out, mode=mode, cross=cross)
            if self.se3_flag:
                self.lin_ori = VecLinear(
                    in_features, nonlinear_out, mode=mode, cross=cross
                )
            self.safe_bound = safe_bound
            self.use_safe_activation = self.safe_bound > 0
            if self.use_safe_activation:
                self.tanh = torch.nn.Hardtanh(-safe_bound, safe_bound)

        self.use_normalization = normalization_method is not None
        if self.use_normalization:
            logging.debug(
                f"Note, the VecActivation layer use {normalization_method}, not scale equivariant"
            )
            self.nrm = normalization_method(in_features)

        self.pre_act_len_bound = pre_act_len_bound

    def forward(self, x):
        # B,C,3,...
        if self.act_mode == "len":
            v_len = x.norm(dim=2, keepdim=True) + EPS
            if self.pre_act_len_bound > 0:
                bounded_v_len = self.pre_act_len_bound * torch.tanh(
                    v_len / self.pre_act_len_bound
                )
                x = x / v_len * bounded_v_len
                v_len = bounded_v_len
            act_len = self.act_func(v_len)
            y = x / v_len * act_len
            return y
        else:
            assert x.shape[2] == 3, "not vector neuron"
            q = x
            k, _ = self.lin_dir(x)
            if self.se3_flag:
                o, _ = self.lin_ori(x)
                q = q - o
                k = k - o

            # 可选：对长度做归一化（非尺度等变）
            if self.use_normalization:
                q_dir = F.normalize(q, dim=2)
                q_len = q.norm(dim=2)  # note: the shape into BN is [B,C,...]
                q_len_normalized = self.nrm(q_len.reshape(x.shape[0], x.shape[1], -1))
                q_len_normalized = q_len_normalized.reshape(q_len.shape)
                q = q_dir * q_len_normalized.unsqueeze(2)

            if self.use_safe_activation:
                # 安全激活：在与 k 的平行分量上施加有界非线性
                qTk = (q * k).sum(dim=2, keepdim=True)
                kTk = (k * k).sum(dim=2, keepdim=True)
                factor = qTk / (kTk + EPS)
                factor = self.tanh(factor)
                q_acted = q + k * (self.act_func(factor) - factor)
            else:
                # 常规非线性：分解为平行/正交分量
                k_dir = F.normalize(k, dim=2)
                q_para_len = (q * k_dir).sum(dim=2, keepdim=True)
                q_orthogonal = q - q_para_len * k_dir
                acted_len = self.act_func(q_para_len)
                q_acted = q_orthogonal + k_dir * acted_len

            if self.se3_flag:
                q_acted = q_acted + o
            return q_acted

class VecMeanPool(nn.Module):
    def __init__(self, pooling_dim=-1, **kwargs) -> None:  # Have dummy args here
        super().__init__()
        self.pooling_dim = pooling_dim

    def forward(self, x, return_weight=False):
        if return_weight:
            return x.mean(self.pooling_dim), None
        else:
            return x.mean(self.pooling_dim)

class VecMaxPool(nn.Module):
    """
    等变的“最大池化”：
    - 通过与一个可学习方向向量的相似度做 soft/hard 选择
    - softmax_factor>0 则做软选择（可微），否则做硬选择
    """
    def __init__(
        self,
        in_features,
        shared_nonlinearity=False,
        mode="so3",
        pooling_dim=-1,
        softmax_factor=-1.0,  # if positive, use softmax, else hard pool not learnable
        cross=False,
    ) -> None:
        super().__init__()

        mode = mode.lower()
        assert mode in ["so3", "se3"], "mode must be so3 or se3"
        self.se3_flag = mode == "se3"
        self.shared_nonlinearity_flag = shared_nonlinearity

        nonlinear_out = 1 if self.shared_nonlinearity_flag else in_features
        if self.se3_flag:
            self.lin_ori = VecLinear(in_features, nonlinear_out, mode=mode, cross=cross)

        assert (
            pooling_dim < 0 or pooling_dim >= 3
        ), "invalid pooling dim, the input should have [B,C,3,...] shape"
        self.pooling_dim = pooling_dim

        self.softmax_factor = softmax_factor
        if self.softmax_factor > 0.0:
            # learnable soft selection
            self.lin_k = VecLinear(in_features, in_features, mode=mode, cross=cross)

    def forward(self, x, return_weight=False):
        # B,C,3,...
        reshape_flag = False
        if x.ndim == 5:  # B,C,3,N,K
            B, C, _, N, K = x.shape
            x = x.permute(0, 3, 1, 2, 4).reshape(B * N, C, 3, K)
            reshape_flag = True
        else:
            B, C, _, N = x.shape

        assert not return_weight or x.ndim == 4, "now don't support 5dim return weight"
        assert x.shape[2] == 3, "not vector neuron"

        q = x
        # get k
        k = x.mean(dim=self.pooling_dim, keepdim=True)
        k, _ = self.lin_k(k)

        if self.se3_flag:
            o, _ = self.lin_ori(k)
            q = q - o
            k = k - o
        k_scale_inv = channel_equi_vec_normalize(k)
        if self.softmax_factor > 0.0:  # Soft max pool
            q_scale_inv = channel_equi_vec_normalize(q)
            sim3_invariant_w = (q_scale_inv * k_scale_inv).mean(dim=2, keepdim=True)
            pooling_w = torch.softmax(
                self.softmax_factor * sim3_invariant_w, dim=self.pooling_dim
            )
            out = (x * pooling_w).sum(self.pooling_dim)
            if reshape_flag:
                out = out.reshape(B, N, C, 3).permute(0, 2, 3, 1)
            if return_weight:
                return out, pooling_w
            else:
                return out
        else:  # hard max pool
            q_para_len = (q * k_scale_inv).sum(dim=2, keepdim=True)
            selection = torch.argmax(q_para_len, dim=self.pooling_dim, keepdim=True)
            _expand_args = [-1] * selection.ndim
            _expand_args[2] = 3
            selection = selection.expand(*_expand_args)
            selected_x = torch.gather(input=x, dim=self.pooling_dim, index=selection)
            selected_x = selected_x.squeeze(self.pooling_dim)
            if reshape_flag:
                selected_x = selected_x.reshape(B, N, C, 3).permute(0, 2, 3, 1)
            if return_weight:
                return selected_x, None
            else:
                return selected_x

class VecLinNormAct(nn.Module):
    # 线性-（可选标量归一化）-非线性 的组合层
    # 返回 (v_act, s_act 或 None)
    def __init__(
        self,
        v_in: int,
        v_out: int,
        act_func,
        s_in=0,
        s_out=0,
        mode="so3",
        shared_nonlinearity=False,
        vs_dir_learnable=True,
        normalization_method=None,
        cross=False,
        safe_bound=-1,
        act_mode="vec",
        pre_act_len_bound=-1.0,
        return_tuple=True,
    ) -> None:
        super().__init__()
        self.scalar_out_flag = s_out > 0
        self.lin = VecLinear(
            v_in,
            v_out,
            s_in,
            s_out,
            mode=mode,
            vec2scalar_dir_learnable=vs_dir_learnable,
            cross=cross,
        )
        self.act = VecActivation(
            v_out,
            act_func,
            shared_nonlinearity,
            mode,
            cross=cross,
            normalization_method=normalization_method,
            safe_bound=safe_bound,
            act_mode=act_mode,
            pre_act_len_bound=pre_act_len_bound,
        )
        self.act_func = act_func
        self.return_tuple = return_tuple

        self.use_s_normalization = normalization_method is not None and s_out > 0
        if self.use_s_normalization:
            self.s_nrm = normalization_method(s_out)
        return

    def forward(self, v, s=None):
        if isinstance(v, tuple) and len(v) > 1:
            v, s = v
        if self.scalar_out_flag:  # hybrid mode
            v_out, s_out = self.lin(v, s)
            v_act = self.act(v_out)
            if self.use_s_normalization:
                s_out = self.s_nrm(s_out)
            s_act = self.act_func(s_out)
            return v_act, s_act
        else:
            v_out, _ = self.lin(v, s)
            v_act = self.act(v_out)
            if self.return_tuple:
                return v_act, None
            else:
                return v_act

class VecResBlock(nn.Module):
    # 向量残差块（可含标量通道）
    # 顺序：VecLinNormAct -> VecLinear -> (可选)VecActivation -> Shortcut 相加
    def __init__(
        self,
        v_in,
        v_out,
        v_hidden,
        act_func,
        s_in=0,
        s_out=0,
        s_hidden=0,
        mode="so3",
        last_activate=True,
        vs_dir_learnable=True,
        cross=False,
        normalization_method=None,
        safe_bound=-1.0,
    ) -> None:
        super().__init__()

        self.last_activate = last_activate
        self.act_func = act_func

        self.s_in_features = s_in
        self.s_out_features = s_out
        self.s_hidden_features = s_hidden

        self.fc0 = VecLinNormAct(
            v_in=v_in,
            v_out=v_hidden,
            s_in=s_in,
            s_out=s_hidden,
            act_func=act_func,
            shared_nonlinearity=False,
            mode=mode,
            vs_dir_learnable=vs_dir_learnable,
            cross=cross,
            normalization_method=normalization_method,
        )

        self.lin1 = VecLinear(
            v_in=v_hidden,
            v_out=v_out,
            s_in=s_hidden,
            s_out=s_out,
            mode=mode,
            vec2scalar_dir_learnable=vs_dir_learnable,
            cross=cross,
        )

        if self.last_activate:
            self.act2 = VecActivation(
                in_features=v_out,
                act_func=act_func,
                shared_nonlinearity=False,
                mode=mode,
                cross=cross,
                normalization_method=normalization_method,
                safe_bound=safe_bound,
            )

        self.shortcut = None if v_in == v_out else VecLinear(v_in, v_out, mode=mode)
        if (
            self.s_in_features > 0
            and self.s_out_features > 0
            and self.s_in_features != self.s_out_features
        ):
            self.s_shortcut = nn.Linear(
                self.s_in_features, self.s_out_features, bias=True
            )
        else:
            self.s_shortcut = None

        self.se3_flag = mode == "se3"
        if self.se3_flag:
            # SE(3) 下做一次“平移项”的抵消
            self.subtract = VecLinear(v_in, v_out, mode="se3")

    def forward(self, v, s=None):
        # strict shape x: [B,C,3,N]; [B,C,N]
        assert v.shape[2] == 3, "vec dim should be at dim [2]"
        if self.s_in_features == 0:
            s = None  # 行为由初始化决定，而非传入参数

        v_net, s_net = self.fc0(v, s)
        dv, ds = self.lin1(v_net, s_net)

        if self.shortcut is not None:
            v_s, _ = self.shortcut(v)
        else:
            v_s = v
        v_out = v_s + dv
        if self.se3_flag:
            subtract, _ = self.subtract(v)
            v_out = v_out - subtract
        if self.last_activate:
            v_out = self.act2(v_out)

        if self.s_shortcut is not None:
            assert s is not None and ds is not None
            s_s = self.s_shortcut(s.transpose(-1, 1)).transpose(-1, 1)
            s_out = s_s + ds
        elif ds is not None:  # s_in == s_out or s_in = 0
            if s is None:
                s_out = ds
            else:
                s_out = s + ds
        else:
            s_out = None

        if s_out is not None and self.last_activate:
            s_out = self.act_func(s_out)
        return v_out, s_out

class PointNetEncoder(nn.Module):
    def __init__(self, h_dim=128, c_dim=128, num_layers=4, **kwargs):
        super().__init__()

        self.h_dim = h_dim
        self.c_dim = c_dim
        self.num_layers = num_layers

        self.pool = meanpool

        act_func = nn.LeakyReLU(negative_slope=0.0, inplace=False)
        self.act = act_func

        self.conv_in = nn.Conv1d(3, h_dim, kernel_size=1)
        self.layers, self.global_layers = nn.ModuleList(), nn.ModuleList()
        for i in range(self.num_layers):
            self.layers.append(nn.Conv1d(h_dim, h_dim, kernel_size=1))
            self.global_layers.append(nn.Conv1d(h_dim * 2, h_dim, kernel_size=1))
        self.conv_out = nn.Conv1d(h_dim * self.num_layers, c_dim, kernel_size=1)

    def forward(self, x, ret_perpoint_feat=False):

        y = self.act(self.conv_in(x))
        feat_list = []
        for i in range(self.num_layers):
            y = self.act(self.layers[i](y))
            y_global = y.max(-1, keepdim=True).values
            y = torch.cat([y, y_global.expand_as(y)], dim=1)
            y = self.act(self.global_layers[i](y))
            feat_list.append(y)
        x = torch.cat(feat_list, dim=1)
        x = self.conv_out(x)

        x_global = x.max(-1).values

        ret = {"global": x_global}
        if ret_perpoint_feat:
            ret["per_point"] = x.transpose(1, 2)

        return ret

        self,

        vec_feature = cond

        if scalar_cond is not None:
            scalar_feature = torch.cat([scalar_feature, scalar_cond], dim=-1)

        if scalar_sample is not None:
            assert hasattr(self, "scalar_fusion_module")
            sample = self.scalar_fusion_module(sample, scalar_sample)

        x = sample
        h = []
        for idx, (resnet, resnet2, downsample) in enumerate(self.down_modules):
            x = resnet(x, vec_feature, scalar_feature)
            x = resnet2(x, vec_feature, scalar_feature)
            h.append(x)
            x = downsample(x)

        for mid_module in self.mid_modules:
            x = mid_module(x, vec_feature, scalar_feature)

        for idx, (resnet, resnet2, upsample) in enumerate(self.up_modules):
            x = torch.cat((x, h.pop()), dim=1)
            x = resnet(x, vec_feature, scalar_feature)
            x = resnet2(x, vec_feature, scalar_feature)
            x = upsample(x)

        if scalar_sample is not None:
            assert hasattr(self, "final_linear")
            x_scalar = self.final_linear(x)[1]
            x_scalar = einops.rearrange(x_scalar, "b t h -> b h t")

        x = self.final_conv(x)
        x = einops.rearrange(x, "b t v h -> b h t v")

        if scalar_sample is not None:
            return x, x_scalar
        else:
            return x, None

        super().__init__()

            out_dim = in_dim

        original_shape = x.shape

        x_flattened = x_flattened.reshape(-1, *x_flattened.shape[2:])
        out = self.conv(x_flattened)

        out = torch.transpose(out.reshape(-1, 3, *out.shape[1:]), 1, 2)

def to_torch(x: Any, device: torch.device | str):
    """
    递归地将 numpy 数组与张量搬到指定 device。
    - dict: 递归到值
    - list/tuple: 递归到元素（保持类型）
    - numpy.ndarray: 转为 torch.tensor 并搬到 device
    - torch.Tensor: 直接搬到 device
    - 其他类型: 原样返回
    """
    if isinstance(x, dict):
        return {k: to_torch(v, device) for k, v in x.items()}
    if isinstance(x, (list, tuple)):
        return type(x)(to_torch(v, device) for v in x)
    if torch.is_tensor(x):
        return x.to(device)
    if isinstance(x, np.ndarray):
        return torch.from_numpy(x).to(device)
    return x

def rotate_around_z(
    points,
    angle_rad=0.0,
    center=np.array([0.0, 0.0, 0.0]),
    scale=np.array([1.0, 1.0, 1.0]),
):
    # 简单的围绕 z 轴旋转与缩放（用于数据增强或可视化）
    assert (len(points.shape) == 1 and len(points) == 3) or points.shape[-1] == 3
    p_shape = points.shape
    points = points.reshape(-1, 3) - center[None]

    # Create the rotation matrix
    cos_theta = np.cos(angle_rad)
    sin_theta = np.sin(angle_rad)
    rotation_matrix = np.array(
        [[cos_theta, -sin_theta, 0], [sin_theta, cos_theta, 0], [0, 0, 1]]
    )

    # Apply the rotation to all points using matrix multiplication
    rotated_points = np.dot(points, rotation_matrix.T) * scale[None] + center[None]
    rotated_points = rotated_points.reshape(p_shape)

    return rotated_points

import numpy as np
import torch


class Normalizer(object):
    def __init__(self, data, symmetric=False, indices=None):
        if isinstance(data, dict):

            self.stats = data
        elif symmetric:

            if indices is None:
                indices = np.arange(data.shape[-1])[None]

            self.stats = {
                "min": torch.zeros([data.shape[-1]]).to(data.device),
                "max": torch.ones([data.shape[-1]]).to(data.device),
            }
            for group in indices:

                max_abs = torch.abs(data[:, group]).max(0)[0].detach()
                limits = torch.ones_like(max_abs) * torch.max(max_abs)
                self.stats["max"][group] = limits
        else:

            mask = torch.zeros([data.shape[-1]]).to(data.device)
            if indices is not None:
                mask[indices.flatten()] += 1
            else:
                mask += 1
            self.stats = {
                "min": data.min(0)[0].detach() * mask,
                "max": data.max(0)[0].detach() * mask + 1.0 * (1 - mask),
            }

    def normalize(self, data):
        nd = len(data.shape)
        target_shape = (1,) * (nd - 1) + (data.shape[-1],)
        dmin = self.stats["min"].reshape(target_shape)
        dmax = self.stats["max"].reshape(target_shape)
        return (data - dmin) / (dmax - dmin + 1e-12)

    def unnormalize(self, data):
        nd = len(data.shape)
        target_shape = (1,) * (nd - 1) + (data.shape[-1],)
        dmin = self.stats["min"].reshape(target_shape)
        dmax = self.stats["max"].reshape(target_shape)
        return data * (dmax - dmin) + dmin

    def state_dict(self):
        return self.stats

    def load_state_dict(self, state_dict):
        self.stats = state_dict

from torch.nn.modules.batchnorm import _BatchNorm


class EMAModel:

    def __init__(
        self,
        model,
        update_after_step=0,
        inv_gamma=1.0,
        power=2 / 3,
        min_value=0.0,
        max_value=0.9999,
    ):

        self.averaged_model = model
        self.averaged_model.eval()
        self.averaged_model.requires_grad_(False)

        self.update_after_step = update_after_step
        self.inv_gamma = inv_gamma
        self.power = power
        self.min_value = min_value
        self.max_value = max_value

        self.decay = 0.0
        self.optimization_step = 0

    def get_decay(self, optimization_step):
        step = max(0, optimization_step - self.update_after_step - 1)
        value = 1 - (1 + step / self.inv_gamma) ** -self.power

        if step <= 0:
            return 0.0

        return max(self.min_value, min(value, self.max_value))

    @torch.no_grad()
    def step(self, new_model):
        self.decay = self.get_decay(self.optimization_step)

        all_dataptrs = set()
        for module, ema_module in zip(

import torch
import torch.nn as nn


class SinusoidalPosEmb(nn.Module):

    def __init__(self, dim):
        super().__init__()
        self.dim = dim

    def forward(self, x):
        device = x.device
        half_dim = self.dim // 2
        emb = math.log(10000) / (half_dim - 1)
        emb = torch.exp(torch.arange(half_dim, device=device) * -emb)
        emb = x[:, None] * emb[None, :]
        emb = torch.cat((emb.sin(), emb.cos()), dim=-1)
        return emb

class ConditionalResidualBlock1D(nn.Module):
    def __init__(
        self,
        in_channels,
        out_channels,
        cond_dim,
        kernel_size=3,
        n_groups=8,
        cond_predict_scale=False,
    ):
        super().__init__()

        self.blocks = nn.ModuleList(
            [
                Conv1dBlock(in_channels, out_channels, kernel_size, n_groups=n_groups),
                Conv1dBlock(out_channels, out_channels, kernel_size, n_groups=n_groups),
            ]
        )

        cond_channels = out_channels
        if cond_predict_scale:
            cond_channels = out_channels * 2
        self.cond_predict_scale = cond_predict_scale
        self.out_channels = out_channels
        self.cond_encoder = nn.Sequential(
            nn.Mish(),
            nn.Linear(cond_dim, cond_channels),
            Rearrange("batch t -> batch t 1"),
        )

        self.residual_conv = (
            nn.Conv1d(in_channels, out_channels, 1)
            if in_channels != out_channels
            else nn.Identity()
        )

    def forward(self, x, cond):
        out = self.blocks[0](x)
        embed = self.cond_encoder(cond)
        if self.cond_predict_scale:
            embed = embed.reshape(embed.shape[0], 2, self.out_channels, 1)
            scale = embed[:, 0, ...]
            bias = embed[:, 1, ...]
            out = scale * out + bias
        else:
            out = out + embed
        out = self.blocks[1](out)
        out = out + self.residual_conv(x)
        return out

            resnet, resnet2 = self.local_cond_encoder
            x = resnet(local_cond, global_feature)
            h_local.append(x)
            x = resnet2(local_cond, global_feature)
            h_local.append(x)

        x = sample

            x = upsample(x)

        x = self.final_conv(x)

        x = einops.rearrange(x, "b t h -> b h t")
        return x

import torch
import torchvision
from torch import nn

from typing import Callable


def get_resnet(name: str, weights=None, **kwargs) -> nn.Module:
    """
    name: resnet18, resnet34, resnet50
    weights: "IMAGENET1K_V1", None
    """
    # Use standard ResNet implementation from torchvision
    func = getattr(torchvision.models, name)
    resnet = func(weights=weights, **kwargs)

    # remove the final fully connected layer
    # for resnet18, the output dim should be 512
    resnet.fc = torch.nn.Identity()
    return resnet


def replace_submodules(
    root_module: nn.Module,
    predicate: Callable[[nn.Module], bool],
    func: Callable[[nn.Module], nn.Module],
) -> nn.Module:
    """
    Replace all submodules selected by the predicate with
    the output of func.

    predicate: Return true if the module is to be replaced.
    func: Return new module to use.
    """
    if predicate(root_module):
        return func(root_module)

    bn_list = [
        k.split(".")
        for k, m in root_module.named_modules(remove_duplicate=True)
        if predicate(m)
    ]
    for *parent, k in bn_list:
        parent_module = root_module
        if len(parent) > 0:
            parent_module = root_module.get_submodule(".".join(parent))
        if isinstance(parent_module, nn.Sequential):
            src_module = parent_module[int(k)]
        else:
            src_module = getattr(parent_module, k)
        tgt_module = func(src_module)
        if isinstance(parent_module, nn.Sequential):
            parent_module[int(k)] = tgt_module
        else:
            setattr(parent_module, k, tgt_module)
    # verify that all modules are replaced
    bn_list = [
        k.split(".")
        for k, m in root_module.named_modules(remove_duplicate=True)
        if predicate(m)
    ]
    assert len(bn_list) == 0
    return root_module


def replace_bn_with_gn(
    root_module: nn.Module, features_per_group: int = 16
) -> nn.Module:
    """
    Relace all BatchNorm layers with GroupNorm.
    """
    replace_submodules(
        root_module=root_module,
        predicate=lambda x: isinstance(x, nn.BatchNorm2d),
        func=lambda x: nn.GroupNorm(
            num_groups=x.num_features // features_per_group, num_channels=x.num_features
        ),
    )
    return root_module

import torch

EPS = 1e-8

def _skew(v):
    # v: (...,3)
    O = torch.zeros_like(v[..., 0])
    vx, vy, vz = v[..., 0], v[..., 1], v[..., 2]
    return torch.stack(
        [
            torch.stack([O, -vz, vy], dim=-1),
            torch.stack([vz, O, -vx], dim=-1),
            torch.stack([-vy, vx, O], dim=-1),
        ],
        dim=-2,
    )

def so3_exp_map(omega):
    """
    Axis-angle to rotation matrix. omega: (..., 3)
    Returns: (..., 3, 3)
    """
    theta = torch.linalg.norm(omega, dim=-1, keepdim=True).clamp(min=EPS)  # (...,1)
    k = omega / theta  # (...,3)
    K = _skew(k)       # (...,3,3)
    I = torch.eye(3, device=omega.device, dtype=omega.dtype).expand(*omega.shape[:-1], 3, 3)
    s = torch.sin(theta)[..., None]  # (...,1,1)
    c = torch.cos(theta)[..., None]
    R = I + s * K + (1 - c) * (K @ K)
    # Handle small-angle numerics by Taylor expansion near 0
    small = (theta.squeeze(-1) < 1e-6)[..., None, None]
    if small.any():
        # Use 1st order: R ≈ I + [omega]_x
        K_omega = _skew(omega)
        R_small = I + K_omega
        R = torch.where(small, R_small, R)
    return R

def so3_log_map(R):
    """
    Rotation matrix to axis-angle. R: (...,3,3)
    Returns: (...,3) axis-angle vector
    """
    # Clamp trace to [-1,3]
    tr = (R[..., 0, 0] + R[..., 1, 1] + R[..., 2, 2]).clamp(-1.0, 3.0)
    cos_theta = (tr - 1.0) / 2.0
    cos_theta = cos_theta.clamp(-1.0, 1.0)
    theta = torch.acos(cos_theta)  # (...,)

    # For general case
    wx = R[..., 2, 1] - R[..., 1, 2]
    wy = R[..., 0, 2] - R[..., 2, 0]
    wz = R[..., 1, 0] - R[..., 0, 1]
    w = torch.stack([wx, wy, wz], dim=-1)  # (...,3)

    # Avoid division by zero
    denom = 2.0 * torch.sin(theta).unsqueeze(-1).clamp(min=EPS)
    v = theta.unsqueeze(-1) * (w / denom)

    # Near zero angle: use first-order approx v ≈ vee(R - R^T)/2
    small = (theta.abs() < 1e-6)[..., None]
    v_small = 0.5 * torch.stack(
        [R[..., 2, 1] - R[..., 1, 2],
         R[..., 0, 2] - R[..., 2, 0],
         R[..., 1, 0] - R[..., 0, 1]], dim=-1
    )
    v = torch.where(small, v_small, v)
    return v

def compose_axis_angles(a, b):
    """
    Compose rotations Exp(a) * Exp(b) = Exp(c), return c.
    a, b: (...,3)
    """
    Ra = so3_exp_map(a)
    Rb = so3_exp_map(b)
    Rc = Ra @ Rb
    return so3_log_map(Rc)

def relative_axis_angle(a_from, a_to):
    """
    Compute relative rotation r s.t. Exp(a_from)^T Exp(a_to) = Exp(r).
    """
    Rf = so3_exp_map(a_from)
    Rt = so3_exp_map(a_to)
    Rrel = Rf.transpose(-1, -2) @ Rt
    return so3_log_map(Rrel)

def vec6d_to_R(eef_pos_6drot):
    """
    Convert 6D rotation representation stored in state (two columns of R) to orthonormal R.
    eef_pos_6drot: (..., 6) with first 3 pos, next two 3D vectors as R[:,0], R[:,1].
    Returns: (..., 3, 3)
    """
    x = eef_pos_6drot[..., 0:3]  # not used here (pos)
    c1 = eef_pos_6drot[..., 3:6]
    c2 = eef_pos_6drot[..., 6:9] if eef_pos_6drot.shape[-1] >= 9 else None
    if c2 is None:
        # Fallback: identity
        I = torch.eye(3, device=eef_pos_6drot.device, dtype=eef_pos_6drot.dtype)
        tgt = I.expand(*eef_pos_6drot.shape[:-1], 3, 3)
        return tgt
    u1 = torch.nn.functional.normalize(c1, dim=-1)
    u2 = c2 - (u1 * c2).sum(-1, keepdim=True) * u1
    u2 = torch.nn.functional.normalize(u2, dim=-1)
    u3 = torch.cross(u1, u2, dim=-1)
    R = torch.stack([u1, u2, u3], dim=-1)  # (...,3,3), columns
    return R

        if len(self.file_names[0].split("_")[-1].split(".")[0][1:]) == 4:
            self.use_four_digit_time = True
        else:
            self.use_four_digit_time = False

        if self.precache:
            self._init_cache()
        else:
            self.cache = dict()

    def __len__(self):
        return len(self.file_names)

    def __getitem__(self, idx):
        fn = self.file_names[idx]
        key_fn = lambda x: "_".join(x.split("/")[-1].split("_")[:-1])
        ep_t = int(fn.split("_")[-1][1:-4])
        start_t = ep_t - (self.obs_horizon - 1)
        end_t = ep_t + self.pred_horizon
        ep_t_list = np.arange(start_t, end_t)
        clipped_ep_t_list = np.clip(ep_t_list, 0, self.ep_length_dict[key_fn(fn)] - 1)

        ret = dict(eef_pose_w=[], eef_pos=[], action=[])
        if self.obs_mode.startswith("pc"):
            ret["pc"] = []
            if len(self.rgb_keys) > 0:
                ret["rgb"] = {k: [] for k in self.rgb_keys}
        else:
            ret["rgb"] = {k: [] for k in self.rgb_keys}
        for t, clipped_t in zip(ep_t_list, clipped_ep_t_list):

            if self.use_four_digit_time:
                fn_t = "_".join(fn.split("_")[:-1] + [f"t{clipped_t:04d}.npz"])
            else:
                fn_t = "_".join(fn.split("_")[:-1] + [f"t{clipped_t:02d}.npz"])

            if t < ep_t:
                keys = ["eef_pose_w"]
                if self.obs_mode == "pc":
                    keys.append("pc")

                    if len(self.rgb_keys) > 0:
                        keys.append("rgb")
                elif self.obs_mode == "rgb":
                    keys.append("rgb")
            elif t == ep_t:
                keys = ["action", "eef_pose_w"]
                if self.obs_mode == "pc":
                    keys.append("pc")
                    if len(self.rgb_keys) > 0:
                        keys.append("rgb")
                elif self.obs_mode == "rgb":
                    keys.append("rgb")
            else:
                keys = ["action"]
            data_t = self._process_data_from_file(fn_t, keys)

            if t >= ep_t:
                ret["action"].append(data_t["action"])

            if t < ep_t or t == ep_t:

                if "eef_pos" in data_t:
                    ret["eef_pos"].append(data_t["eef_pos"])

                if "eef_pose_w" in data_t:
                    ret["eef_pose_w"].append(data_t["eef_pose_w"])
                if self.obs_mode.startswith("pc") and "pc" in data_t:
                    ret["pc"].append(data_t["pc"])

                if "rgb" in data_t and "rgb" in ret:
                    for k in self.rgb_keys:
                        ret["rgb"][k].append(data_t["rgb"][k])

        if len(ret["eef_pos"]) > 0:
            ret["eef_pos"] = np.array(ret["eef_pos"])
        if len(ret["eef_pose_w"]) > 0:
            ret["eef_pose_w"] = np.array(ret["eef_pose_w"])
        if len(ret["action"]) > 0:
            ret["action"] = np.array(ret["action"])
        if self.obs_mode.startswith("pc"):
            ret["pc"] = np.array(ret["pc"])
        if "rgb" in ret:
            for k in self.rgb_keys:
                ret["rgb"][k] = np.array(ret["rgb"][k])

        assert len(ret["action"]) == self.pred_horizon

        if (
            "eef_pos" in ret
            and isinstance(ret["eef_pos"], np.ndarray)
            and ret["eef_pos"].ndim > 0
        ):
            assert len(ret["eef_pos"]) == self.obs_horizon
        if (
            "eef_pose_w" in ret
            and isinstance(ret["eef_pose_w"], np.ndarray)
            and ret["eef_pose_w"].ndim > 0
        ):
            assert len(ret["eef_pose_w"]) == self.obs_horizon
        if "pc" in ret and isinstance(ret["pc"], np.ndarray) and ret["pc"].ndim > 0:
            assert len(ret["pc"]) == self.obs_horizon
        if "rgb" in ret:
            for k in self.rgb_keys:
                assert k in ret["rgb"], f"Expected view key {k} in returned rgb dict"
                assert len(ret["rgb"][k]) == self.obs_horizon, (
                    f"Expected {self.obs_horizon} frames for view {k}, "
                    f"got {len(ret['rgb'][k])}"
                )

        return ret

    def _init_cache(self):

        self.cache = dict()

        keys_to_keep = ["eef_pose_w", "action"]
        if self.obs_mode.startswith("pc"):
            keys_to_keep.append("pc")
        if self.obs_mode == "rgb" and self.cache_rgb:
            keys_to_keep.extend(self.rgb_keys)

        for fn in tqdm(self.file_names, desc="Building dataset cache"):
            with np.load(fn) as data:
                self.cache[fn] = dict()

                if "eef_pos" in data.files:
                    self.cache[fn]["eef_pos"] = data["eef_pos"].astype(np.float32)
                if "eef_pose_w" in data.files:
                    self.cache[fn]["eef_pose_w"] = data["eef_pose_w"].astype(np.float32)

                self.cache[fn]["action"] = data["action"].astype(np.float32)

                if "pc" in keys_to_keep and "pc" in data.files:
                    self.cache[fn]["pc"] = data["pc"].astype(np.float32)
                if self.obs_mode == "rgb" and self.cache_rgb:
                    for k in self.rgb_keys:
                        if k in data.files:
                            self.cache[fn][k] = data[k]

    def _process_data_from_file(self, fn, keys):
        entry = self.cache.get(fn, {})
        file_data = None

        def ensure_file():
            nonlocal file_data
            if file_data is None:
                file_data = np.load(fn)
            return file_data

        ret = {}

        if "eef_pos" in keys:
            if "eef_pos" in entry:
                eef_pos = entry["eef_pos"].astype(np.float32)
            else:
                fd = ensure_file()
                if "eef_pos" not in fd.files:

                    eef_pos = None
                else:
                    eef_pos = fd["eef_pos"].astype(np.float32)
            if eef_pos is not None:
                eef_pos = eef_pos.reshape(self.num_eef, -1)[:, : self.eef_dim]
                ret["eef_pos"] = eef_pos

        if "eef_pose_w" in keys:
            if "eef_pose_w" in entry:
                eef6 = entry["eef_pose_w"].astype(np.float32)
            else:
                fd = ensure_file()
                if "eef_pose_w" not in fd.files:
                    eef6 = None
                else:
                    eef6 = fd["eef_pose_w"].astype(np.float32)
            if eef6 is not None:

                eef6 = eef6.reshape(self.num_eef, -1)
                ret["eef_pose_w"] = eef6

        if "action" in keys:
            if "action" in entry:
                action = entry["action"].astype(np.float32)
            else:
                fd = ensure_file()
                if "action" not in fd.files:
                    raise KeyError(f"Missing 'action' in NPZ '{fn}'.")
                action = fd["action"].astype(np.float32)
            ret["action"] = action.flatten().astype(np.float32)

        if "pc" in keys:
            if "pc" in entry:
                xyz = entry["pc"].astype(np.float32)
            else:
                fd = ensure_file()
                if "pc" not in fd.files:
                    raise KeyError(
                        f"Missing 'pc' in NPZ '{fn}'. "
                        f"Re-collect demos with point clouds enabled (FactoryIL runner: --save_pc) "
                        f"and re-run `factoryil.tools.convert_to_npz`."
                    )
                xyz = fd["pc"].astype(np.float32)

            N = xyz.shape[0]
            K = self.num_points

            if N == 0:

                ret["pc"] = xyz.astype(np.float32)
            else:
                if self.mode == "train" and self.shuffle_pc:

                    if N >= K:

                        choice = np.random.choice(N, K, replace=False)
                    else:

                        choice = np.random.choice(N, K, replace=True)
                    xyz = xyz[choice, :]
                else:

                    if N > K:
                        step = max(1, N // K)
                        xyz = xyz[::step, :][:K, :]
                    elif N < K:

                        choice = np.random.choice(N, K, replace=True)
                        xyz = xyz[choice, :]

                ret["pc"] = xyz.astype(np.float32)

        if "rgb" in keys:

            rgb_dict = {}
            for k in self.rgb_keys:
                if k in entry:
                    rgb = entry[k]
                else:
                    fd = ensure_file()

                    rgb = fd[k]
                rgb_dict[k] = rgb
            ret["rgb"] = rgb_dict

        if file_data is not None:
            file_data.close()

        return ret

#!/usr/bin/env python3
from __future__ import annotations

"""Convert FactoryIL episode folders into per-frame `.npz` files for IL training.

Input layout (written by `factoryil.runner` via `EpisodeStager`):
  episodeXX/
    rgb_cam/0000.png
    depth_cam/0000.npy
    rgb_tactile_left/0000.png
    rgb_tactile_right/0000.png
    pc/0000.npy                      (optional)
    timestamps_robot.csv             (frame,is_first,is_last,proprio,action,eef_pose_w)
    env_id.txt / env_origin.txt      (optional)

Output layout:
  <out_dir>/episodeXX_tYYYY.npz

Each NPZ may contain (depending on what exists on disk):
  - rgb_cam: (H, W, 3) uint8
  - depth_cam: (H, W) float32 (raw depth)
  - rgb_tactile_left/right: (H, W, 3) uint8
  - pc: (P, 3) float32 (optionally sampled to `--pc_num_points`)
  - action: (7,) float32 (pads/clamps; last dim reserved for gripper, set to 0.0)
  - eef_pose_w: (10,) float32 (pads/clamps; last dim reserved for gripper, set to 0.0)
"""

import argparse
import concurrent.futures
import csv
import glob
import multiprocessing
import os
from collections import OrderedDict
from typing import Dict, List, Optional, Sequence, Set, Tuple

import numpy as np
from PIL import Image


def _read_uint8_png(path: str) -> np.ndarray:
    with Image.open(path) as im:
        im = im.convert("RGB")
        return np.asarray(im, dtype=np.uint8)


def _read_depth_npy(path: str) -> np.ndarray:
    arr = np.load(path)
    if arr.ndim == 3 and arr.shape[0] == 1:
        arr = arr[0]
    return arr.astype(np.float32, copy=False)


def _parse_semicolon_floats(s: str) -> List[float]:
    s = (s or "").strip()
    if not s:
        return []
    out: List[float] = []
    for p in [p.strip() for p in s.split(";") if p.strip()]:
        try:
            out.append(float(p))
        except Exception:
            continue
    return out


def _list_episode_dirs(root: str) -> List[str]:
    return sorted([d for d in glob.glob(os.path.join(root, "episode*")) if os.path.isdir(d)])


def _read_csv_rows(csv_path: str) -> List[Dict[str, str]]:
    if not os.path.exists(csv_path):
        raise FileNotFoundError(f"CSV not found: {csv_path}")
    rows: List[Dict[str, str]] = []
    with open(csv_path, "r", newline="", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        for row in reader:
            rows.append(row)
    return rows


def _frame_has_any_modalities(ep_dir: str, frame: int) -> bool:
    fname = f"{frame:04d}"
    if os.path.isfile(os.path.join(ep_dir, "rgb_cam", f"{fname}.png")):
        return True
    if os.path.isfile(os.path.join(ep_dir, "depth_cam", f"{fname}.npy")):
        return True
    if os.path.isfile(os.path.join(ep_dir, "rgb_tactile_left", f"{fname}.png")):
        return True
    if os.path.isfile(os.path.join(ep_dir, "rgb_tactile_right", f"{fname}.png")):
        return True
    if os.path.isfile(os.path.join(ep_dir, "pc", f"{fname}.npy")):
        return True
    return False


def _sample_rows_fixed(n_rows: int, n: int, rng: np.random.Generator):
    """Return indices to sample `n_rows` rows to exactly `n` rows."""
    if n is None or n <= 0:
        return slice(None)
    if n_rows <= 0:
        return np.arange(0, dtype=np.int64)
    if n_rows >= n:
        idx = rng.choice(n_rows, size=n, replace=False)
    else:
        idx = rng.choice(n_rows, size=n, replace=True)
    return idx.astype(np.int64)


def _collect_modalities_for_frame(ep_dir: str, frame: int, include_images: bool) -> Dict[str, object]:
    fname = f"{frame:04d}"
    out: Dict[str, object] = {}
    if not include_images:
        return out

    rgb_path = os.path.join(ep_dir, "rgb_cam", f"{fname}.png")
    if os.path.isfile(rgb_path):
        try:
            out["rgb_cam"] = _read_uint8_png(rgb_path)
        except Exception:
            pass

    depth_path = os.path.join(ep_dir, "depth_cam", f"{fname}.npy")
    if os.path.isfile(depth_path):
        try:
            out["depth_cam"] = _read_depth_npy(depth_path)
        except Exception:
            pass

    tl_path = os.path.join(ep_dir, "rgb_tactile_left", f"{fname}.png")
    if os.path.isfile(tl_path):
        try:
            out["rgb_tactile_left"] = _read_uint8_png(tl_path)
        except Exception:
            pass
    tr_path = os.path.join(ep_dir, "rgb_tactile_right", f"{fname}.png")
    if os.path.isfile(tr_path):
        try:
            out["rgb_tactile_right"] = _read_uint8_png(tr_path)
        except Exception:
            pass

    return out


def _ensure_action7(action_vals: Sequence[float]) -> np.ndarray:
    arr = np.asarray(action_vals, dtype=np.float32).reshape(-1)
    if arr.size >= 7:
        out = arr[:7].astype(np.float32, copy=True)
        out[-1] = 0.0
        return out
    pad = np.zeros((7 - arr.size,), dtype=np.float32)
    out = np.concatenate([arr, pad], axis=0).astype(np.float32)
    out[-1] = 0.0
    return out


def _ensure_eef_pose10(eef_vals: Sequence[float]) -> np.ndarray:
    arr = np.asarray(eef_vals, dtype=np.float32).reshape(-1)
    if arr.size >= 10:
        out = arr[:10].astype(np.float32, copy=True)
        out[-1] = 0.0
        return out
    pad = np.zeros((10 - arr.size,), dtype=np.float32)
    out = np.concatenate([arr, pad], axis=0).astype(np.float32)
    out[-1] = 0.0
    return out


def _get_already_converted_episodes(out_dir: str) -> Set[str]:
    pattern = os.path.join(out_dir, "episode*_t*.npz")
    files = glob.glob(pattern)
    episodes: Set[str] = set()
    for f in files:
        base = os.path.basename(f)
        if "_t" in base:
            episodes.add(base.split("_t", 1)[0])
    return episodes


def _convert_single_episode(
    ep_dir: str,
    out_dir: str,
    pc_num_points: int,
    seed: int,
    compress: bool,
    pc_only: bool,
) -> None:
    ep_name = os.path.basename(os.path.normpath(ep_dir))
    csv_path = os.path.join(ep_dir, "timestamps_robot.csv")

    try:
        rows = _read_csv_rows(csv_path)
    except Exception as e:
        print(f"[Skip] {ep_name}: failed to read CSV: {e}")
        return

    rng = np.random.default_rng(seed)

    for row in rows:
        try:
            frame_idx = int(float(row.get("frame", "0")))
        except Exception:
            continue

        if not _frame_has_any_modalities(ep_dir, frame_idx):
            continue

        save_dict: Dict[str, object] = OrderedDict()
        save_dict.update(_collect_modalities_for_frame(ep_dir, frame_idx, include_images=not pc_only))

        pc_path = os.path.join(ep_dir, "pc", f"{frame_idx:04d}.npy")
        if os.path.isfile(pc_path):
            try:
                pc_xyz = np.load(pc_path).astype(np.float32, copy=False)
                if pc_xyz.ndim == 2 and pc_xyz.shape[1] >= 3:
                    pc_xyz = pc_xyz[:, :3]
                    if pc_num_points and pc_num_points > 0:
                        idx = _sample_rows_fixed(pc_xyz.shape[0], pc_num_points, rng)
                        pc_xyz = pc_xyz[idx, :] if not isinstance(idx, slice) else pc_xyz
                    save_dict["pc"] = pc_xyz.astype(np.float32, copy=False)
            except Exception:
                pass

        action_vals = _parse_semicolon_floats(row.get("action", ""))
        eef_vals = _parse_semicolon_floats(row.get("eef_pose_w", ""))
        save_dict["action"] = _ensure_action7(action_vals)
        save_dict["eef_pose_w"] = _ensure_eef_pose10(eef_vals)

        out_name = f"{ep_name}_t{frame_idx:04d}.npz"
        out_path = os.path.join(out_dir, out_name)
        if compress:
            np.savez_compressed(out_path, **save_dict)
        else:
            np.savez(out_path, **save_dict)


def convert_episodes(
    episodes_root: str,
    out_dir: str,
    pc_num_points: int,
    pc_random_seed: int,
    num_workers: int = 0,
    compress: bool = True,
    pc_only: bool = False,
    start_from: Optional[str] = None,
) -> None:
    os.makedirs(out_dir, exist_ok=True)
    episode_dirs = _list_episode_dirs(episodes_root)
    if not episode_dirs:
        raise FileNotFoundError(f"No episode* directories under: {episodes_root}")

    start_from_index: Optional[int] = None
    if start_from:
        for idx, ep_dir in enumerate(episode_dirs):
            if os.path.basename(os.path.normpath(ep_dir)) == start_from:
                start_from_index = idx
                break

    already = _get_already_converted_episodes(out_dir)

    tasks: List[Tuple[str, str, int, int, bool, bool]] = []
    for i, ep_dir in enumerate(episode_dirs):
        ep_name = os.path.basename(os.path.normpath(ep_dir))
        force = start_from_index is not None and i >= start_from_index

        if not force and ep_name in already:
            continue

        if force and ep_name in already:
            for f in glob.glob(os.path.join(out_dir, f"{ep_name}_t*.npz")):
                try:
                    os.remove(f)
                except OSError:
                    pass

        tasks.append((ep_dir, out_dir, pc_num_points, pc_random_seed + i, compress, pc_only))

    if not tasks:
        print("[convert_to_npz] Nothing to do (all episodes already converted).")
        return

    max_workers = int(num_workers) if int(num_workers) > 0 else multiprocessing.cpu_count()
    print(f"[convert_to_npz] Converting {len(tasks)} episodes with {max_workers} worker(s)...")

    with concurrent.futures.ProcessPoolExecutor(max_workers=max_workers) as ex:
        futs = [ex.submit(_convert_single_episode, *t) for t in tasks]
        for fut in concurrent.futures.as_completed(futs):
            fut.result()

    print("[convert_to_npz] Done.")


def main() -> None:
    ap = argparse.ArgumentParser(description="Convert FactoryIL episodes to per-frame NPZ files.")
    ap.add_argument("--episodes_root", "-i", required=True, help="Root directory containing episodeXX folders.")

        start_from=args.start_from,
    )


if __name__ == "__main__":

def _ensure_len(history: List[Any], target_len: int) -> List[Any]:
    """Left-pad history by repeating the first element until `target_len`."""
    if target_len <= 0:
        return []
    if len(history) >= target_len:
        return history[-target_len:]
    if len(history) == 0:
        raise ValueError("Cannot pad an empty history.")
    first = history[0]
    return [first for _ in range(target_len - len(history))] + history


def _chw01_to_hwc_u8(chw01: torch.Tensor) -> np.ndarray:
    """CHW float tensor in [0,1] -> HWC uint8 numpy."""
    t = torch.clamp(chw01, 0.0, 1.0)
    return (t.permute(1, 2, 0) * 255.0).round().to(torch.uint8).cpu().numpy()


def _resolve_latest_ckpt(path: str) -> str:
    """If `path` is a directory, resolve to the latest ckpt_best*.pth inside it."""
    p = Path(path)
    if not p.exists():
        return path
    if p.is_file():
        return str(p)
    # directory
    candidates = list(p.rglob("ckpt_best*.pth"))
    if not candidates:
        # fall back to any .pth
        candidates = list(p.rglob("*.pth"))
    if not candidates:
        raise FileNotFoundError(f"No checkpoint found under directory: {path}")
    latest = max(candidates, key=lambda x: x.stat().st_mtime)
    return str(latest)

                "ILPolicyAgent requires OmegaConf (omegaconf). Install it (e.g. `pip install -e .[il]`) "

            ) from e

        # Locate configs relative to the vendored policy package.
        try:
            import imitation_learning.policies as pol_pkg
        except Exception as e:  # pragma: no cover
            raise ImportError(
                "Could not import `imitation_learning.policies`. "
                "Make sure `src/third_party` is on the python path (see pyproject.toml)."
            ) from e

        pkg_dir = Path(getattr(pol_pkg, "__file__", "")).resolve().parent
        config_dir = pkg_dir / "configs"

        cfg_path = Path(self.cfg_name)
        if cfg_path.suffix == ".yaml" and cfg_path.exists():
            cfg_file = cfg_path
        else:
            cfg_file = config_dir / f"{self.cfg_name}.yaml"
        if not cfg_file.exists():
            raise FileNotFoundError(f"IL config not found: {cfg_file}")

        # NOTE: the policy configs use Hydra-style defaults (e.g. `defaults: - base`).
        # For inference inside FactoryIL we do a minimal "compose" by merging
        # `base.yaml` + the selected config.
        base_file = config_dir / "base.yaml"
        if not base_file.exists():
            raise FileNotFoundError(f"Missing base policy config: {base_file}")
        cfg_base = OmegaConf.load(str(base_file))
        cfg_spec = OmegaConf.load(str(cfg_file))
        cfg = OmegaConf.merge(cfg_base, cfg_spec)
        # Best-effort overrides for evaluation/inference.
        try:
            cfg.mode = "eval"
        except Exception:
            pass
        try:
            cfg.device = str(device)
        except Exception:
            pass
        try:
            cfg.use_wandb = False
        except Exception:
            pass

        # Instantiate the underlying IL agent.
        try:
            from imitation_learning.policies.utils.misc import get_agent
        except Exception as e:  # pragma: no cover
            raise ImportError("Failed importing IL policy utilities from vendored code.") from e

        AgentCls = get_agent(cfg.agent.agent_name)
        self._agent = AgentCls(cfg)
        self._agent.load_snapshot(self.ckpt_path)
        self._agent.train(False)

        # Resolve observation spec from agent.
        self.obs_mode: str = str(getattr(self._agent, "obs_mode", "pc"))
        self.obs_horizon = int(getattr(self._agent, "obs_horizon", 2))
        self.pred_horizon = int(getattr(self._agent, "pred_horizon", 8))
        self.ac_horizon = int(getattr(self._agent, "ac_horizon", 1))
        self.num_eef = int(getattr(self._agent, "num_eef", 1))
        self.dof = int(getattr(self._agent, "dof", 7))

        # RGB view keys for rgb-mode agents.
        self.rgb_keys: List[str] = list(getattr(self._agent, "rgb_keys", []))

        # Per-env state is allocated in reset().
        self._env_state: List[_PerEnvILState] = []

    # ------------------------------------------------------------------ #
    # FactoryAgent hooks
    # ------------------------------------------------------------------ #

    def _reset_internal_state(self) -> None:
        self._env_state = []
        for _ in range(int(self.num_envs)):
            self._env_state.append(
                _PerEnvILState(
                    state_hist=[],
                    pc_hist=[],
                    rgb_hist={k: [] for k in self.rgb_keys},
                    pred_buf=None,
                    pred_idx=0,
                    steps_left=0,
                )
            )

    def _reset_episode_internal(self, env_idx: int) -> None:
        if env_idx < 0 or env_idx >= len(self._env_state):
            return
        st = self._env_state[env_idx]
        st.state_hist.clear()
        st.pc_hist.clear()
        for k in st.rgb_hist:
            st.rgb_hist[k].clear()
        st.pred_buf = None
        st.pred_idx = 0
        st.steps_left = 0

    def _alloc_action_batch(self, env) -> torch.Tensor:
        action_dim = int(env.unwrapped.cfg.action_space)
        return torch.zeros((int(self.num_envs), action_dim), device=env.device, dtype=torch.float32)

    def _zero_action_for_env(self, env, env_idx: int) -> torch.Tensor:  # noqa: ARG002
        action_dim = int(env.unwrapped.cfg.action_space)
        a = torch.zeros((action_dim,), device=self.device, dtype=torch.float32)
        if action_dim > 6:
            a[6] = -1.0
        return a

    def _compute_action_for_env(self, env_idx: int, env, t_step: int) -> torch.Tensor:  # noqa: ARG002
        # Unused because we override `act` vectorized over envs.
        return self._zero_action_for_env(env, env_idx)

    # ------------------------------------------------------------------ #
    # Main policy inference entrypoint
    # ------------------------------------------------------------------ #

    @torch.no_grad()
    def act(self, obs_batch, done_mask, env, t_step: int):  # noqa: ARG002
        n = int(self.num_envs or len(obs_batch))
        if self.num_envs != n:
            self.reset(n)

        done = np.asarray(done_mask, dtype=bool) if done_mask is not None else np.zeros((n,), dtype=bool)

        action_dim_env = int(env.unwrapped.cfg.action_space)
        out = torch.zeros((n, action_dim_env), device=env.device, dtype=torch.float32)

        for i in range(n):
            if done[i]:
                out[i] = self._zero_action_for_env(env, i)
                continue

            obs_i = obs_batch[i] or {}
            st = self._env_state[i]

            # ------------------------------------------------------------------
            # 1) Update per-env observation histories (pre-step obs_t).
            # ------------------------------------------------------------------
            state_t = obs_i.get("state", None)
            if state_t is None:
                raise KeyError("ILPolicyAgent expected obs_i['state'] (EEF pose) to be present.")

            if torch.is_tensor(state_t):
                state_np = state_t.detach().to("cpu").numpy().astype(np.float32).reshape(-1)
            else:
                state_np = np.asarray(state_t, dtype=np.float32).reshape(-1)
            # Convert 9D -> 10D by appending a constant 0.0 (gripper).
            if state_np.size == 9:
                state_np = np.concatenate([state_np, np.zeros((1,), dtype=np.float32)], axis=0)
            if state_np.size < 10:
                pad = np.zeros((10 - state_np.size,), dtype=np.float32)
                state_np = np.concatenate([state_np, pad], axis=0)
            state_np = state_np[:10].astype(np.float32, copy=False).reshape(self.num_eef, 10)
            st.state_hist.append(state_np)

            if self.obs_mode.startswith("pc"):
                pc_t = obs_i.get("pc", None)
                if pc_t is not None:
                    if torch.is_tensor(pc_t):
                        pc_np = pc_t.detach().to("cpu").numpy().astype(np.float32)
                    else:
                        pc_np = np.asarray(pc_t, dtype=np.float32)
                    if pc_np.ndim == 2 and pc_np.shape[1] >= 3:
                        pc_np = pc_np[:, :3]
                    st.pc_hist.append(pc_np)

            if self.obs_mode == "rgb":
                rgb_views = obs_i.get("rgb", {}) or {}
                if not self.rgb_keys:
                    # First call: infer view keys from provided observation.
                    self.rgb_keys = list(rgb_views.keys())
                    st.rgb_hist = {k: [] for k in self.rgb_keys}
                for k in self.rgb_keys:
                    v = rgb_views.get(k, None)
                    if v is None:
                        continue
                    if torch.is_tensor(v):
                        st.rgb_hist[k].append(_chw01_to_hwc_u8(v))
                    else:
                        st.rgb_hist[k].append(np.asarray(v))

            # ------------------------------------------------------------------
            # 2) Run inference when the current predicted sequence is exhausted.
            # ------------------------------------------------------------------
            need_new = (
                st.pred_buf is None
                or st.steps_left <= 0
                or st.pred_idx >= int(self.pred_horizon)
            )
            if need_new:
                state_seq = _ensure_len(st.state_hist, int(self.obs_horizon))
                agent_obs: Dict[str, Any] = {"state": np.stack(state_seq, axis=0)}
                if self.obs_mode.startswith("pc"):
                    if len(st.pc_hist) == 0:
                        raise RuntimeError(
                            "ILPolicyAgent is in pc mode but did not receive any 'pc' observations yet. "
                            "Did you enable --save_pc and set up segmentation?"
                        )
                    pc_seq = _ensure_len(st.pc_hist, int(self.obs_horizon))
                    agent_obs["pc"] = pc_seq  # list of arrays
                else:
                    rgb_dict: Dict[str, np.ndarray] = {}
                    for k in self.rgb_keys:
                        if len(st.rgb_hist.get(k, [])) == 0:
                            continue
                        seq = _ensure_len(st.rgb_hist[k], int(self.obs_horizon))
                        rgb_dict[k] = np.stack(seq, axis=0)
                    agent_obs["rgb"] = rgb_dict

                pred = self._agent.act(agent_obs, return_dict=False, debug=False)
                st.pred_buf = np.asarray(pred, dtype=np.float32)
                st.pred_idx = 0
                st.steps_left = min(int(self.ac_horizon), int(st.pred_buf.shape[0]))

            # ------------------------------------------------------------------
            # 3) Execute one step from the predicted sequence.
            # ------------------------------------------------------------------
            assert st.pred_buf is not None
            if st.pred_idx >= st.pred_buf.shape[0]:
                exec_one = st.pred_buf[-1]
            else:
                exec_one = st.pred_buf[st.pred_idx]

            st.pred_idx += 1
            st.steps_left -= 1

            exec_one = np.asarray(exec_one, dtype=np.float32).reshape(-1)
            # Predicted vector is [num_eef*dof]; take the first EEF and first 6 DOFs.
            if exec_one.size >= self.dof:
                exec_eef = exec_one[: self.dof]
            else:
                pad = np.zeros((self.dof - exec_one.size,), dtype=np.float32)
                exec_eef = np.concatenate([exec_one, pad], axis=0)

            a = np.zeros((action_dim_env,), dtype=np.float32)
            a[:6] = exec_eef[:6]
            if action_dim_env > 6:
                a[6] = -1.0  # keep gripper closed during IL rollouts
            a = np.clip(a, -1.0, 1.0)
            out[i] = torch.as_tensor(a, device=env.device, dtype=torch.float32)

        return out

def _write_eef_debug_artifacts(episode_dir: str) -> None:
    """Write `eef_debug.csv` and `eef_debug.png` into an episode directory."""
    csv_path = os.path.join(episode_dir, "timestamps_robot.csv")
    if not os.path.exists(csv_path):

        return

        return

    try:
        out_csv = os.path.join(episode_dir, "eef_debug.csv")
        with open(out_csv, "w", encoding="utf-8", newline="") as f:
            w = csv.writer(f)
            w.writerow(["step", "eef_z", "eef_yx", "eef_yy", "eef_yz"])

        draw = ImageDraw.Draw(img)
        try:
            font = ImageFont.load_default()
        except Exception:
            font = None

        def _draw_panel(
            rect: tuple[int, int, int, int],
            series: list[tuple[str, list[float], tuple[int, int, int]]],
            title: str,
        ) -> None:
            x0, y0, x1, y1 = rect
            draw.rectangle(rect, outline=(0, 0, 0), width=1)

            # X scaling from step indices (use min/max).
            xmin = float(min(steps))

            for _, vals, _ in series:

            ymax = float(max(all_vals))

            pad = 0.05 * (ymax - ymin)
            ymin -= pad
            ymax += pad

            def _pt(x: float, y: float) -> tuple[int, int]:
                px = x0 + int(round((x - xmin) / (xmax - xmin) * (x1 - x0)))
                py = y1 - int(round((y - ymin) / (ymax - ymin) * (y1 - y0)))
                return px, py

        bot = (margin, margin + panel_h + gap, margin + panel_w, margin + panel_h + gap + panel_h)

def _tensor_like_to_list(x):
    """Convert torch tensors / numpy arrays / lists to plain python lists."""
    try:
        if torch.is_tensor(x):
            return x.detach().to("cpu").tolist()
    except Exception:
        pass
    try:
        import numpy as np  # local import (optional)

        if isinstance(x, np.ndarray):
            return x.tolist()
    except Exception:
        pass
    return x

def _atomic_json_dump(path: str, payload: dict) -> None:
    """Write JSON atomically to avoid partial files if the process is killed."""
    os.makedirs(os.path.dirname(path), exist_ok=True)
    tmp_path = f"{path}.tmp"
    with open(tmp_path, "w", encoding="utf-8") as f:
        json.dump(payload, f, indent=2, sort_keys=False)
    os.replace(tmp_path, path)

@torch.no_grad()
def _run_peg_insert_with_agent(args, env, agent) -> dict | None:
    """Shared peg-insertion rollout for any agent."""
    device = env.device
    num_envs: int = env.unwrapped.num_envs
    action_dim: int = env.unwrapped.cfg.action_space

    mode = str(getattr(args, "mode", "collect")).lower().strip()
    if mode not in ("collect", "evaluate"):
        mode = "collect"

    dataset_root, success_root, failed_root, staging_root = _alloc_roots_for_expert(args)
    wrote_cam_params = False

    zero_action = torch.zeros((num_envs, action_dim), device=device)
    discard_timeouts = bool(getattr(args, "discard_timeouts", False))
    # Evaluate mode always counts failures; keep them on disk for inspection.
    if mode == "evaluate":
        discard_timeouts = False

    target_episodes = int(args.episodes)
    total_success = 0
    episodes_ended = 0
    episodes_succeeded = 0

    batch_idx = 0

    def _should_stop() -> bool:
        if mode == "collect":
            return total_success >= target_episodes
        # evaluate: run exactly N total episodes (success + fail)
        return episodes_ended >= target_episodes

    def _accept_episode() -> bool:
        """Whether we should save/count another episode given the current budget."""
        if mode == "evaluate":
            return episodes_ended < target_episodes
        # collect: once we have enough successes, stop saving anything else
        return total_success < target_episodes

    while not _should_stop():
        batch_idx += 1
        if mode == "collect":
            print(
                f"[FACTORYIL] Starting batch {batch_idx} | "
                f"successes={total_success}/{target_episodes}",
                flush=True,
            )
        else:
            print(
                f"[FACTORYIL] Starting batch {batch_idx} | "
                f"episodes={episodes_ended}/{target_episodes} successes={total_success}",
                flush=True,
            )

        # ------------------------------------------------------------------ #
        # 1) Episode start: reset buffers.                                   #
        # ------------------------------------------------------------------ #
        raw_obs, _ = env.reset(seed=None)
        if not wrote_cam_params:
            # Persist camera params once per dataset root for offline video tooling.
            _maybe_write_camera_params_json(
                env, data_root=dataset_root, failed_root=failed_root
            )
            wrote_cam_params = True
        # IMPORTANT: each env.reset() starts a fresh episode batch, so reset the
        # agent's per-episode/buffer state as well (IL agents keep history).
        try:
            agent.reset(num_envs)
        except Exception:
            pass

        env._compute_intermediate_values()
        step_idx = 0

        # One stager per env.
        stagers = [EpisodeStager(staging_root) for _ in range(num_envs)]
        env_done = torch.zeros(num_envs, dtype=torch.bool, device=device)
        env_first = torch.ones(num_envs, dtype=torch.bool, device=device)

        # Determine step cap from env's configured episode length.
        max_len = int(getattr(env.unwrapped, "max_episode_length", 0))
        if max_len <= 0:
            max_len = 1000
        step_cap = max_len

        modality_cfg = C.ModalityConfig(
            use_rgb=True,
            use_depth=True,
            use_tactile=True,
            use_pc=False,
            stack_rgb=C.STACK_RGB,
        )

        while True:
            try:
                env.unwrapped.sim.render()
            except Exception:
                pass

            if step_idx % 10 == 0:
                active_envs = int((~env_done).sum().item())
                if mode == "collect":
                    print(
                        f"[FACTORYIL] batch={batch_idx} step={step_idx} "
                        f"active_envs={active_envs} successes={total_success}/{target_episodes}",
                        flush=True,
                    )
                else:
                    print(
                        f"[FACTORYIL] batch={batch_idx} step={step_idx} "
                        f"active_envs={active_envs} episodes={episodes_ended}/{target_episodes} successes={total_success}",
                        flush=True,
                    )

            # ------------------------------------------------------------------
            # 2) Pre-step observations in world frames.                        #
            # ------------------------------------------------------------------
            obs_pre = build_obs_dict(
                env,
                raw_obs,
                modality_cfg,
                device,
            )

            rgb_pre = obs_pre.get("rgb_camera", None)
            depth_pre = obs_pre.get("depth_camera", None)
            proprio_pre = obs_pre["proprio"]
            eef_pose_pre = obs_pre["eef_pose_w"]

            # Tactile images.
            try:
                tl_pre, tr_pre = get_tactile_batch(
                    env,
                    tact_h=C.TACT_H,
                    tact_w=C.TACT_W,
                    device=device,
                )
            except Exception:
                tl_pre, tr_pre = (None, None)

            # Optional per-frame point clouds: env/world frames.
            pc_env_pre = [None for _ in range(num_envs)]
            # Compute per-frame PCs when:
            #   - explicit --save_pc is requested (for logging), or
            #   - the agent itself uses point clouds (obs_mode startswith "pc").
            wants_pc = bool(getattr(args, "save_pc", False))
            if hasattr(agent, "obs_mode"):
                try:
                    if str(getattr(agent, "obs_mode", "")).startswith("pc"):
                        wants_pc = True
                except Exception:
                    pass
            if wants_pc:
                try:
                    pcs_fixed_world, _ = get_object_pointclouds(
                        env,
                        max_points=int(getattr(args, "pc_max_points", 4096)),
                        device=device,
                    )
                except Exception:
                    pcs_fixed_world = None
                if pcs_fixed_world is not None:
                    try:
                        env_origins = env.scene.env_origins.to(device)
                        for i in range(num_envs):
                            pc_i = pcs_fixed_world[i]
                            if pc_i is None or pc_i.numel() == 0:
                                continue
                            # Per-env frame: subtract this env's origin.
                            pc_env_i = pc_i.to(device) - env_origins[i].unsqueeze(0)
                            pc_env_pre[i] = pc_env_i
                    except Exception:
                        pc_env_pre = [None for _ in range(num_envs)]

            # ------------------------------------------------------------------
            # 3) Agent control in world frame.                                 #
            # ------------------------------------------------------------------
            # Build per-env rich observation dicts; agents decide which
            # modalities to use (ILPolicyAgent keeps its own history).
            is_pc_agent = bool(
                hasattr(agent, "obs_mode")
                and str(getattr(agent, "obs_mode", "")).startswith("pc")
            )

            obs_batch = []
            for i in range(num_envs):
                obs_i = {}

                # State for IL / generic agents: always world-frame.
                state_i = eef_pose_pre[i]
                obs_i["state"] = state_i

                # Optional point cloud (env frame) for PC-mode IL agents.
                pc_i = None
                if is_pc_agent or wants_pc:
                    pc_i = pc_env_pre[i]
                if pc_i is not None:
                    obs_i["pc"] = pc_i

                # RGB / tactile views (as RGB images).
                rgb_views = {}
                if rgb_pre is not None and rgb_pre[i] is not None:
                    rgb_views["rgb_cam"] = rgb_pre[i]
                if tl_pre is not None and tl_pre[i] is not None:
                    rgb_views["rgb_tactile_left"] = tl_pre[i]
                if tr_pre is not None and tr_pre[i] is not None:
                    rgb_views["rgb_tactile_right"] = tr_pre[i]
                if rgb_views:
                    obs_i["rgb"] = rgb_views

                # Low-level proprio / poses kept for potential future agents.
                obs_i["proprio"] = proprio_pre[i]
                obs_i["eef_pose_w"] = eef_pose_pre[i]

                obs_batch.append(obs_i)

            done_mask_np = env_done.detach().to("cpu").numpy().astype(bool)
            actions = agent.act(obs_batch, done_mask_np, env, t_step=step_idx)

            # ------------------------------------------------------------------
            # 4) Logging at time t.                                            #
            # ------------------------------------------------------------------
            for i in range(num_envs):
                if bool(env_done[i].item()):
                    continue
                stagers[i].add_frame(
                    rgb_chw=(rgb_pre[i] if rgb_pre is not None else None),
                    depth_chw=(depth_pre[i] if depth_pre is not None else None),
                    tact_left_chw=(tl_pre[i] if tl_pre is not None else None),
                    tact_right_chw=(tr_pre[i] if tr_pre is not None else None),
                    proprio_1d=proprio_pre[i],
                    action_1d=actions[i],
                    eef_pose_1d=eef_pose_pre[i],
                    is_first=bool(env_first[i].item()),
                    is_last=False,
                    pc_xyz=pc_env_pre[i],
                )
                env_first[i] = False

            # Step environment.
            raw_obs, _, terminated, truncated, _ = env.step(actions)
            # Success signal.
            succ = success_mask(env)
            ended_flags = (terminated.to(torch.bool) | truncated.to(torch.bool))

            # For envs that have just succeeded or ended, append a final zero-action frame.
            if bool((succ | ended_flags).any().item()):
                obs_post = build_obs_dict(
                    env,
                    raw_obs,
                    modality_cfg,
                    device,
                )
                rgb_post = obs_post.get("rgb_camera", None)
                depth_post = obs_post.get("depth_camera", None)
                proprio_post = obs_post["proprio"]
                eef_pose_post = obs_post["eef_pose_w"]

                try:
                    tl_post, tr_post = get_tactile_batch(
                        env,
                        tact_h=C.TACT_H,
                        tact_w=C.TACT_W,
                        device=device,
                    )
                except Exception:
                    tl_post, tr_post = (None, None)

                pc_env_post = [None for _ in range(num_envs)]
                if getattr(args, "save_pc", False):
                    try:
                        pcs_fixed_world_post, _ = get_object_pointclouds(
                            env,
                            max_points=int(getattr(args, "pc_max_points", 4096)),
                            device=device,
                        )
                    except Exception:
                        pcs_fixed_world_post = None
                    if pcs_fixed_world_post is not None:
                        try:
                            env_origins = env.scene.env_origins.to(device)
                            for i in range(num_envs):
                                pc_i = pcs_fixed_world_post[i]
                                if pc_i is None or pc_i.numel() == 0:
                                    continue
                                pc_env_i = pc_i.to(device) - env_origins[i].unsqueeze(0)
                                pc_env_post[i] = pc_env_i
                        except Exception:
                            pc_env_post = [None for _ in range(num_envs)]

                for i in range(num_envs):
                    if bool(env_done[i].item()):
                        continue

                    is_succ = bool(succ[i].item())
                    ended = bool(ended_flags[i].item())

                    if is_succ:
                        if not _accept_episode():
                            stagers[i].discard()
                            env_done[i] = True
                            continue
                        stagers[i].add_frame(
                            rgb_chw=(rgb_post[i] if rgb_post is not None else None),
                            depth_chw=(depth_post[i] if depth_post is not None else None),
                            tact_left_chw=(tl_post[i] if tl_post is not None else None),
                            tact_right_chw=(tr_post[i] if tr_post is not None else None),
                            proprio_1d=proprio_post[i],
                            action_1d=zero_action[i],
                            eef_pose_1d=eef_pose_post[i],
                            is_first=False,
                            is_last=True,
                            pc_xyz=pc_env_post[i],
                        )
                        origin_i = None
                        try:
                            origin_i = env.scene.env_origins[i]
                        except Exception:
                            origin_i = None
                        dest = stagers[i].finalize(
                            success_root, env_id=i, env_origin=origin_i
                        )
                        if _DEBUG_EEF_PLOTS and isinstance(agent, ExpertAgent):
                            _write_eef_debug_artifacts(dest)

                        total_success += 1
                        episodes_ended += 1
                        episodes_succeeded += 1
                        env_done[i] = True

                        sr = episodes_succeeded / max(1, episodes_ended)
                        print(
                            f"[FACTORYIL] SUCCESS env={i} -> {dest} | "
                            f"success_rate={episodes_succeeded}/{episodes_ended} ({sr:.2%})",
                            flush=True,
                        )

                    elif ended:
                        if not _accept_episode():
                            stagers[i].discard()
                            env_done[i] = True
                            continue
                        if discard_timeouts:
                            stagers[i].discard()
                        else:
                            stagers[i].add_frame(
                                rgb_chw=(rgb_post[i] if rgb_post is not None else None),
                                depth_chw=(depth_post[i] if depth_post is not None else None),
                                tact_left_chw=(tl_post[i] if tl_post is not None else None),
                                tact_right_chw=(tr_post[i] if tr_post is not None else None),
                                proprio_1d=proprio_post[i],
                                action_1d=zero_action[i],
                                eef_pose_1d=eef_pose_post[i],
                                is_first=False,
                                is_last=True,
                                pc_xyz=(pc_env_post[i] if pc_env_post is not None else None),
                            )
                            origin_i = None
                            try:
                                origin_i = env.scene.env_origins[i]
                            except Exception:
                                origin_i = None
                            dest = stagers[i].finalize(
                                failed_root, env_id=i, env_origin=origin_i
                            )
                            if _DEBUG_EEF_PLOTS and isinstance(agent, ExpertAgent):
                                _write_eef_debug_artifacts(dest)

                        episodes_ended += 1
                        env_done[i] = True
                        sr = episodes_succeeded / max(1, episodes_ended)
                        if discard_timeouts:
                            print(
                                f"[FACTORYIL] TIMEOUT env={i} -> discarded | "
                                f"success_rate={episodes_succeeded}/{episodes_ended} ({sr:.2%})",
                                flush=True,
                            )
                        else:
                            print(
                                f"[FACTORYIL] TIMEOUT env={i} -> {dest} | "
                                f"success_rate={episodes_succeeded}/{episodes_ended} ({sr:.2%})",
                                flush=True,
                            )

            step_idx += 1

            if _should_stop():
                break
            if bool(env_done.all().item()):
                break
            if step_idx >= step_cap:
                # Force all unfinished envs to be marked as failed for this batch.
                for i in range(num_envs):
                    if not bool(env_done[i].item()):
                        if not _accept_episode():
                            stagers[i].discard()
                            env_done[i] = True
                            continue
                        if discard_timeouts:
                            stagers[i].discard()
                            dest = None
                        else:
                            origin_i = None
                            try:
                                origin_i = env.scene.env_origins[i]
                            except Exception:
                                origin_i = None
                            dest = stagers[i].finalize(
                                failed_root, env_id=i, env_origin=origin_i
                            )
                        episodes_ended += 1
                        env_done[i] = True
                        sr = episodes_succeeded / max(1, episodes_ended)
                        if discard_timeouts:
                            print(
                                f"[FACTORYIL] STEP_CAP TIMEOUT env={i} -> discarded | "
                                f"success_rate={episodes_succeeded}/{episodes_ended} ({sr:.2%})",
                                flush=True,
                            )
                        else:
                            print(
                                f"[FACTORYIL] STEP_CAP TIMEOUT env={i} -> {dest} | "
                                f"success_rate={episodes_succeeded}/{episodes_ended} ({sr:.2%})",
                                flush=True,
                            )
                break

        # Discard any partially written episodes (should be none, but keep robust).
        for i in range(num_envs):
            if not bool(env_done[i].item()):
                stagers[i].discard()

    print(
        f"[FACTORYIL DONE] mode={mode} successes_saved={total_success} "
        f"episodes_ended={episodes_ended} | out={dataset_root}",
        flush=True,
    )
    if mode == "evaluate":
        failures = max(0, int(episodes_ended) - int(episodes_succeeded))
        sr = float(episodes_succeeded) / float(max(1, episodes_ended))
        summary = {
            "mode": "evaluate",
            "episodes_requested": int(target_episodes),
            "episodes_total": int(episodes_ended),
            "successes": int(episodes_succeeded),
            "failures": int(failures),
            "success_rate": float(sr),
            "args": _args_to_json_dict(args),
            "data_dir": str(dataset_root),
            "success_dir": str(success_root),
            "failed_dir": str(failed_root),
        }
        # Persist alongside the saved episodes for experiment tooling.
        try:
            _atomic_json_dump(os.path.join(dataset_root, "eval_summary.json"), summary)
        except Exception:
            pass
        return summary
    return None


@torch.no_grad()
def run(args, env) -> None:
    """Top-level runner selecting agent and task."""
    print("[FACTORYIL] Starting FactoryIL runner...", flush=True)
    device = env.device
    mode = str(getattr(args, "mode", "collect")).lower().strip()
    if mode not in ("collect", "evaluate"):
        mode = "collect"
    if mode == "evaluate":
        # Machine-readable: print args as JSON at the beginning of evaluation.
        try:
            print(json.dumps(_args_to_json_dict(args)), flush=True)
        except Exception:
            pass

    # Instantiate agent based on CLI.
    agent_name = getattr(args, "agent", "expert")
    if agent_name == "expert":
        agent = ExpertAgent(device=device, expert_cfg=args)
    elif agent_name == "replay":
        if not getattr(args, "episode_dir", None):
            raise ValueError("--episode_dir is required when --agent=replay")
        agent = ReplayAgent(
            device=device,
            episode_dir=args.episode_dir,
        )
    elif agent_name == "il":
        if not getattr(args, "il_config", None):
            raise ValueError("--il_config is required when --agent=il")
        if not getattr(args, "il_ckpt", None):
            raise ValueError("--il_ckpt is required when --agent=il")
        # Optional dependency: IL is only required when actually used.
        from factoryil.agents.il_agents import ILPolicyAgent  # noqa: E402

        agent = ILPolicyAgent(
            il_cfg=args.il_config,
            il_checkpoint=args.il_ckpt,
            device=device,
            args=args,
        )
        # Optional ac_horizon override from CLI (clamped to pred_horizon).
        ac_h_override = getattr(args, "ac_horizon", None)
        if ac_h_override is not None:
            try:
                ac_h_candidate = int(ac_h_override)
                if ac_h_candidate > 0:
                    agent.ac_horizon = min(ac_h_candidate, agent.pred_horizon)
            except Exception:
                pass
    else:
        raise NotImplementedError(
            f"Agent type '{agent_name}' is not yet supported in FactoryIL runner."
        )

    num_envs: int = env.unwrapped.num_envs
    agent.reset(num_envs)

    if args.task != "peg_insert":
        raise NotImplementedError(
            f"Unified runner currently supports only peg_insert, got task={args.task!r}."
        )

    summary = _run_peg_insert_with_agent(args, env, agent)
    if mode == "evaluate" and isinstance(summary, dict):
        # Machine-readable: emit evaluation summary as JSON.
        try:
            print(json.dumps(summary), flush=True)
        except Exception:
            pass

def parse_args() -> argparse.Namespace:
    parser = build_factoryil_runner_parser()
    args = parser.parse_args()
    return args


def main() -> None:
    args = parse_args()

    # Launch Isaac Sim / Isaac Lab.
    app_launcher = AppLauncher(vars(args))
    simulation_app = app_launcher.app

    # Delayed imports that require simulation to be running.
    from factoryil.env.factory_env import FactoryILEnv  # noqa: E402
    from factoryil.env.factory_task_cfg import (  # noqa: E402
        FactoryILPegInsertCfg,
        FactoryILGearMeshCfg,
        FactoryILNutThreadCfg,

    # Configure environment.
    if args.task == "peg_insert":
        env_cfg = FactoryILPegInsertCfg()
    elif args.task == "gear_mesh":
        env_cfg = FactoryILGearMeshCfg()
    elif args.task == "nut_thread":
        env_cfg = FactoryILNutThreadCfg()
    else:
        raise ValueError(f"Unknown task '{args.task}'")
    env_cfg.scene.num_envs = int(args.num_envs)
    # Timeout: either explicit timesteps (preferred for deterministic datasets)
    # or a seconds-based time limit.
    timeout_steps = getattr(args, "timeout_steps", None)
    if timeout_steps is not None:
        # One env step corresponds to dt * decimation seconds.
        step_dt_s = float(env_cfg.sim.dt) * float(getattr(env_cfg, "decimation", 1))
        env_cfg.episode_length_s = float(int(timeout_steps)) * step_dt_s
    else:
        env_cfg.episode_length_s = float(args.timeout_s)
    try:
        env_cfg.sim.render_interval = 1
    except Exception:
        pass

    # Apply initialization noise from CLI (zero by default for deterministic runs).
    if hasattr(args, "rand_eef") and float(args.rand_eef) != 0.0:
        val = abs(float(args.rand_eef))
        env_cfg.eef_init_pos_noise = [val, val, 0.0]
    if hasattr(args, "rand_peg") and float(args.rand_peg) != 0.0:
        val = abs(float(args.rand_peg))
        env_cfg.held_init_pos_noise = [val, 0.0, 0.0]
    if hasattr(args, "rand_hole") and float(args.rand_hole) != 0.0:

        env_cfg.fixed_init_pos_noise = [val, val, 0.0]

    env = FactoryILEnv(env_cfg)

    try:
        run(args, env)
    except Exception as e:
        print(f"Error: {e}")
    finally:
        try:
            env.close()
        except Exception:
            pass
        try:
            simulation_app.close()
        except Exception:
            pass


if __name__ == "__main__":
    main()

from __future__ import annotations

from typing import Dict, Optional

import torch

from ... import config as C
from .vision import get_rgb_and_depth


def extract_proprio(raw_obs) -> torch.Tensor:
    """Extract the proprioception vector from the env `raw_obs`."""
    if isinstance(raw_obs, dict):
        v = raw_obs.get("policy", None)
        if v is None:
            # Best-effort fallback: take the first entry.
            v = next(iter(raw_obs.values()))
    else:
        v = raw_obs
    if not torch.is_tensor(v):
        v = torch.as_tensor(v, dtype=torch.float32)
    return v


def extract_eef_pose_w(raw_obs) -> torch.Tensor:
    """Extract 9D EEF pose [pos(3), rot_cols(6)] in env/world frame from `raw_obs`."""
    if isinstance(raw_obs, dict):
        v = raw_obs.get("eef_pose_w", None)
        if v is None:
            # Best-effort fallback: take the first entry.
            v = next(iter(raw_obs.values()))
    else:
        v = raw_obs
    if not torch.is_tensor(v):
        v = torch.as_tensor(v, dtype=torch.float32)
    return v


@torch.no_grad()
def build_obs_dict(env, raw_obs, modality: C.ModalityConfig, device: str) -> Dict[str, Optional[torch.Tensor]]:
    """Build a standardized observation dict for FactoryIL.

    Returned keys (subset depending on `modality`):
    - **proprio**: (N, P)
    - **eef_pose_w**: (N, 9)  [pos(3), rot_cols(6)] in env/world frame
    - **rgb_camera**: (N, 3*stack, H, W) float in [0,1]
    - **depth_camera**: (N, 1, H, W) float (meters)
    """
    obs: Dict[str, Optional[torch.Tensor]] = {}

    proprio = extract_proprio(raw_obs).to(device=device).float()
    eef_pose_w = extract_eef_pose_w(raw_obs).to(device=device).float()
    obs["proprio"] = proprio
    obs["eef_pose_w"] = eef_pose_w

    if modality.use_rgb or modality.use_depth:
        rgb_t, depth_t = get_rgb_and_depth(
            env,
            rgb_h=C.RGB_H,
            rgb_w=C.RGB_W,
            stack=int(modality.stack_rgb),
            want_rgb=bool(modality.use_rgb),
            want_depth=bool(modality.use_depth),
            device=device,
        )
        if modality.use_rgb:
            obs["rgb_camera"] = rgb_t
        if modality.use_depth:
            obs["depth_camera"] = depth_t

    return obs


__all__ = ["extract_proprio", "extract_eef_pose_w", "build_obs_dict"]

from __future__ import annotations

from typing import Tuple

import torch
import torch.nn.functional as F

from factoryil.utils.utils import fetch_tactile_images


@torch.no_grad()
def get_tactile_batch(env, tact_h: int, tact_w: int, device: str) -> Tuple[torch.Tensor, torch.Tensor]:
    """Return (left, right) tactile RGB images as NCHW float tensors in [0, 1]."""
    tacts = fetch_tactile_images(env)
    left_np = tacts["gsmini_left"]["tactile_rgb"]
    right_np = tacts["gsmini_right"]["tactile_rgb"]

    left = torch.as_tensor(left_np, device=device, dtype=torch.float32)
    right = torch.as_tensor(right_np, device=device, dtype=torch.float32)

    # NHWC uint8 -> NCHW float
    left = left.permute(0, 3, 1, 2) / 255.0
    right = right.permute(0, 3, 1, 2) / 255.0

    # Correct orientation: left 90° clockwise, right 90° counter-clockwise
    left = torch.rot90(left, k=-1, dims=(-2, -1))
    right = torch.rot90(right, k=1, dims=(-2, -1))

    if (left.shape[-2] != tact_h) or (left.shape[-1] != tact_w):
        left = F.interpolate(left, size=(tact_h, tact_w), mode="bilinear", align_corners=False)
        right = F.interpolate(right, size=(tact_h, tact_w), mode="bilinear", align_corners=False)

    return left.contiguous(), right.contiguous()


__all__ = ["get_tactile_batch"]

def _ensure_dir(path: str) -> None:
    os.makedirs(path, exist_ok=True)


def _to_uint8_image(arr: np.ndarray) -> np.ndarray:
    """Convert an array-like image to uint8 HWC if possible."""
    arr = np.asarray(arr)
    if arr.dtype != np.uint8:
        m = float(arr.max()) if arr.size else 1.0
        if m <= 1.0 + 1e-6:
            arr = arr * 255.0
        arr = np.clip(arr, 0, 255).astype(np.uint8)
    # If CHW, convert to HWC
    if arr.ndim == 3 and arr.shape[0] in (1, 3, 4) and arr.shape[-1] not in (3, 4):
        arr = np.transpose(arr, (1, 2, 0))
    return arr


def fetch_tactile_images(env) -> Dict[str, Dict[str, np.ndarray]]:
    """Fetch GelSight Mini tactile sensor outputs as uint8 numpy arrays.

    Expected env structure (FactoryILEnv):
    - `env.unwrapped.scene.sensors["gsmini_left"].data.output["tactile_rgb"]`
    - `env.unwrapped.scene.sensors["gsmini_right"].data.output["tactile_rgb"]`

    Returns:
      dict with keys:
        - "gsmini_left":  {"tactile_rgb": (N,H,W,3) uint8, "height_map": (N,H,W) uint8 (if present)}
        - "gsmini_right": {"tactile_rgb": (N,H,W,3) uint8, "height_map": (N,H,W) uint8 (if present)}
    """
    out: Dict[str, Dict[str, np.ndarray]] = {"gsmini_left": {}, "gsmini_right": {}}
    unwrapped = getattr(env, "unwrapped", env)

    def _as_numpy(x):
        """Convert common Isaac/TacEx outputs to numpy.

        GelSight sensors typically output torch tensors (often on CUDA).
        Converting a CUDA tensor via `np.asarray()` raises, which previously
        caused tactile to be silently dropped by callers.
        """
        if x is None:
            return None
        try:
            if torch.is_tensor(x):
                return x.detach().to("cpu").numpy()
        except Exception:
            pass

        # Handle lists/tuples of tensors/arrays.
        if isinstance(x, (list, tuple)) and len(x) > 0:
            try:
                if torch.is_tensor(x[0]):
                    return torch.stack([t.detach().to("cpu") for t in x], dim=0).numpy()
            except Exception:
                pass

        return np.asarray(x)

    def _get_sensor(name: str):
        try:
            return unwrapped.scene.sensors[name]
        except Exception as e:  # pragma: no cover
            raise RuntimeError(f"Sensor {name!r} not found in env.scene.sensors") from e

    left = _get_sensor("gsmini_left")
    right = _get_sensor("gsmini_right")

    left_rgb = left.data.output.get("tactile_rgb", None)
    right_rgb = right.data.output.get("tactile_rgb", None)
    if left_rgb is None or right_rgb is None:
        raise RuntimeError("GelSight sensors did not produce 'tactile_rgb' output.")

    left_rgb = _as_numpy(left_rgb)
    right_rgb = _as_numpy(right_rgb)
    if left_rgb is None or right_rgb is None:
        raise RuntimeError("GelSight sensors produced no tactile output.")
    out["gsmini_left"]["tactile_rgb"] = np.stack(
        [_to_uint8_image(left_rgb[i]) for i in range(left_rgb.shape[0])], axis=0
    )
    out["gsmini_right"]["tactile_rgb"] = np.stack(
        [_to_uint8_image(right_rgb[i]) for i in range(right_rgb.shape[0])], axis=0
    )

    # height_map is optional.
    left_hm = left.data.output.get("height_map", None)
    right_hm = right.data.output.get("height_map", None)
    if left_hm is not None:
        left_hm = _as_numpy(left_hm)
        out["gsmini_left"]["height_map"] = np.stack(
            [_to_uint8_image(left_hm[i]) for i in range(left_hm.shape[0])], axis=0
        )
    if right_hm is not None:
        right_hm = _as_numpy(right_hm)
        out["gsmini_right"]["height_map"] = np.stack(
            [_to_uint8_image(right_hm[i]) for i in range(right_hm.shape[0])], axis=0
        )

    return out


def create_look_at_quat(eye: torch.Tensor, target: torch.Tensor, up_axis: torch.Tensor) -> torch.Tensor:
    """Compute a quaternion (w,x,y,z) so a frame at `eye` looks at `target`.

    This is a small helper used in some debugging/visualization utilities.
    """
    forward_vec = F.normalize(eye - target, p=2, dim=-1)
    right_vec = F.normalize(torch.cross(up_axis.expand_as(forward_vec), forward_vec, dim=-1), p=2, dim=-1)
    up_vec = torch.cross(forward_vec, right_vec, dim=-1)

    rot_matrix = torch.stack((right_vec, up_vec, forward_vec), dim=-1)

    w = 0.5 * torch.sqrt(
        1.0 + rot_matrix[..., 0, 0] + rot_matrix[..., 1, 1] + rot_matrix[..., 2, 2]
    ).unsqueeze(-1)
    x = (0.5 / w) * (rot_matrix[..., 2, 1] - rot_matrix[..., 1, 2]).unsqueeze(-1)
    y = (0.5 / w) * (rot_matrix[..., 0, 2] - rot_matrix[..., 2, 0]).unsqueeze(-1)
    z = (0.5 / w) * (rot_matrix[..., 1, 0] - rot_matrix[..., 0, 1]).unsqueeze(-1)

    return torch.cat((w, x, y, z), dim=-1)

#!/usr/bin/env python
from __future__ import annotations

"""Episode staging / dataset writing utilities for FactoryIL.

This module writes simulator-native observations/actions to disk (no extra frame transforms).

On-disk layout for a single episode directory:
  episodeXX/
    rgb_cam/0000.png
    depth_cam/0000.npy (+ optional 0000.png visualization)
    rgb_tactile_left/0000.png
    rgb_tactile_right/0000.png
    pc/0000.npy                        (optional)
    timestamps_robot.csv
    env_id.txt                         (optional)
    env_origin.txt                     (optional)
"""

import os
from typing import Optional

import numpy as np
import torch
from PIL import Image


def _alloc_next_episode_dir(root: str) -> str:
    """Return a new episode directory path filling the lowest missing index."""
    os.makedirs(root, exist_ok=True)
    existing = [
        d
        for d in os.listdir(root)
        if os.path.isdir(os.path.join(root, d)) and d.startswith("episode")
    ]
    used: set[int] = set()
    for d in existing:
        try:
            idx = int(d.replace("episode", ""))
            used.add(idx)
        except Exception:
            continue
    next_idx = 0
    while next_idx in used:
        next_idx += 1
    return os.path.join(root, f"episode{next_idx:02d}")


def _tensor_to_uint8_img(chw: torch.Tensor) -> np.ndarray:
    """Convert CHW float tensor in [0,1] to HWC uint8 numpy image."""
    t = chw
    if t.ndim != 3:
        raise ValueError(f"Expected CHW tensor, got shape {tuple(t.shape)}")
    t = torch.clamp(t, 0.0, 1.0)
    return (t.permute(1, 2, 0) * 255.0).round().to(torch.uint8).cpu().numpy()


class EpisodeStager:
    """Per-env staging writer that writes frames to a temp directory."""

    def __init__(self, staging_root: str):
        import csv
        import uuid

        self.tmp_dir = os.path.join(staging_root, f"env_{uuid.uuid4().hex[:8]}")
        self.rgb_dir = os.path.join(self.tmp_dir, "rgb_cam")
        self.depth_dir = os.path.join(self.tmp_dir, "depth_cam")
        self.tl_dir = os.path.join(self.tmp_dir, "rgb_tactile_left")
        self.tr_dir = os.path.join(self.tmp_dir, "rgb_tactile_right")
        self.pc_dir = os.path.join(self.tmp_dir, "pc")
        for d in (self.tmp_dir, self.rgb_dir, self.depth_dir, self.tl_dir, self.tr_dir, self.pc_dir):
            os.makedirs(d, exist_ok=True)

        self.csv_path = os.path.join(self.tmp_dir, "timestamps_robot.csv")
        self.csv_file = open(self.csv_path, "w", newline="", encoding="utf-8")
        self.csv_writer = csv.writer(self.csv_file)
        self.csv_writer.writerow(
            ["frame", "is_first", "is_last", "proprio", "action", "eef_pose_w"]
        )

        self.frame_idx = 0
        self.closed = False

    def add_frame(
        self,
        rgb_chw: Optional[torch.Tensor],
        depth_chw: Optional[torch.Tensor],
        tact_left_chw: Optional[torch.Tensor],
        tact_right_chw: Optional[torch.Tensor],
        proprio_1d: torch.Tensor,
        action_1d: torch.Tensor,
        eef_pose_1d: torch.Tensor,
        is_first: bool,
        is_last: bool,
        pc_xyz: Optional[torch.Tensor] = None,
        **_: object,
    ) -> None:
        """Append a frame to the staged episode.

        Extra keyword args are ignored for backwards compatibility.
        """
        if self.closed:
            return

        fname = f"{self.frame_idx:04d}"

        if rgb_chw is not None:
            try:
                Image.fromarray(_tensor_to_uint8_img(rgb_chw)).save(os.path.join(self.rgb_dir, f"{fname}.png"))
            except Exception:
                pass

        # Depth (raw .npy + optional normalized .png for quick inspection).
        depth_hw = None
        if depth_chw is not None:
            try:
                depth_np = depth_chw.detach().to("cpu").numpy()
                if depth_np.ndim == 3 and depth_np.shape[0] == 1:
                    depth_hw = depth_np[0]
                elif depth_np.ndim == 2:
                    depth_hw = depth_np
                else:
                    depth_hw = np.squeeze(depth_np)
                depth_hw = depth_hw.astype(np.float32, copy=False)
                np.save(os.path.join(self.depth_dir, f"{fname}.npy"), depth_hw)
            except Exception:
                depth_hw = None

            if depth_hw is not None:
                try:
                    finite = np.isfinite(depth_hw) & (depth_hw > 0.0)
                    if np.any(finite):
                        dmin = float(np.nanpercentile(depth_hw[finite], 5.0))
                        dmax = float(np.nanpercentile(depth_hw[finite], 95.0))
                        if not np.isfinite(dmin):
                            dmin = 0.0
                        if not np.isfinite(dmax) or dmax <= dmin:
                            dmax = dmin + 1e-6
                        norm = np.clip((depth_hw - dmin) / (dmax - dmin), 0.0, 1.0)
                    else:
                        norm = np.zeros_like(depth_hw, dtype=np.float32)
                    Image.fromarray((norm * 255.0).astype(np.uint8)).save(
                        os.path.join(self.depth_dir, f"{fname}.png")
                    )
                except Exception:
                    pass

        if tact_left_chw is not None:
            try:
                Image.fromarray(_tensor_to_uint8_img(tact_left_chw)).save(os.path.join(self.tl_dir, f"{fname}.png"))
            except Exception:
                pass
        if tact_right_chw is not None:
            try:
                Image.fromarray(_tensor_to_uint8_img(tact_right_chw)).save(os.path.join(self.tr_dir, f"{fname}.png"))
            except Exception:
                pass

        if pc_xyz is not None:
            try:
                np.save(
                    os.path.join(self.pc_dir, f"{fname}.npy"),
                    pc_xyz.detach().to("cpu").numpy().astype(np.float32, copy=False),
                )
            except Exception:
                pass

        proprio_str = ";".join(f"{float(x):.6f}" for x in proprio_1d.detach().cpu().tolist())
        action_str = ";".join(f"{float(x):.6f}" for x in action_1d.detach().cpu().tolist())
        eef_pose_str = ";".join(f"{float(x):.6f}" for x in eef_pose_1d.detach().cpu().tolist())

        self.csv_writer.writerow(
            [self.frame_idx, int(is_first), int(is_last), proprio_str, action_str, eef_pose_str]
        )
        self.frame_idx += 1

    def finalize(self, dataset_root: str, env_id: int, env_origin: Optional[torch.Tensor] = None) -> str:
        """Move staged data to a new episode directory under `dataset_root`."""
        if self.closed:
            return ""

        self.csv_file.flush()
        self.csv_file.close()
        self.closed = True

        import shutil

        dest = _alloc_next_episode_dir(dataset_root)
        os.makedirs(os.path.dirname(dest), exist_ok=True)
        shutil.move(self.tmp_dir, dest)

        # Save env id.
        try:
            with open(os.path.join(dest, "env_id.txt"), "w", encoding="utf-8") as f:
                f.write(f"{int(env_id)}\n")
        except Exception:
            pass

        # Save env origin (for aligning world coordinates).
        if env_origin is not None:
            try:
                origin_np = env_origin.detach().to("cpu").numpy().astype(float).tolist()
                with open(os.path.join(dest, "env_origin.txt"), "w", encoding="utf-8") as f:
                    f.write(f"{origin_np[0]} {origin_np[1]} {origin_np[2]}\n")
            except Exception:
                pass

        return dest

    def discard(self) -> None:
        if self.closed:
            return
        try:
            self.csv_file.close()
        except Exception:
            pass
        self.closed = True
        try:
            import shutil

            shutil.rmtree(self.tmp_dir, ignore_errors=True)
        except Exception:
            pass


def success_mask(env) -> torch.Tensor:
    """Per-env success boolean for peg insertion (peg inserted & centered w.r.t. hole)."""
    task = env.cfg.task
    fixed_cfg = getattr(task, "fixed_asset_cfg", None)
    fixed_height = float(getattr(fixed_cfg, "height", 0.0) or 0.0)

    fixed_pos = env._fixed_asset.data.root_pos_w - env.scene.env_origins  # (N, 3)
    held_pos = env._held_asset.data.root_pos_w - env.scene.env_origins  # (N, 3)

    xy_tol = float(getattr(task, "xy_success_tol_m", 0.0025))
    xy_dist = torch.linalg.vector_norm(held_pos[:, 0:2] - fixed_pos[:, 0:2], dim=1)
    xy_ok = xy_dist < xy_tol

    # Insertion depth: how far peg base is below hole base.
    z_disp = held_pos[:, 2] - fixed_pos[:, 2]  # < 0 means below.
    success_threshold = float(getattr(task, "success_threshold", 0.04))
    height_threshold = fixed_height * success_threshold
    z_ok = z_disp < height_threshold

    return xy_ok & z_ok
